222// For the original promise scheme polyfill:
223// If a set of interceptors might fail asynchronously, wrap the main
224// promise in a race with the cancellation latch.
225// If not, just return the main promise.
226template <bool X>
227struct RaceAsyncCompletion;
228
229template <>
230struct RaceAsyncCompletion<false> {
231 template <typename Promise>
232 static Promise Run(Promise x, void*) {
233 return x;
234 }
235};
236
237template <>
238struct RaceAsyncCompletion<true> {
239 template <typename Promise>
240 static auto Run(Promise x, Latch<ServerMetadataHandle>* latch) {
241 return Race(latch->Wait(), std::move(x));
242 }
243};

1122// Base class for promise-based channel filters.
1123// Eventually this machinery will move elsewhere (the interception logic will
1124// move directly into the channel stack, and so filters will just directly
1125// derive from `ChannelFilter`)
1126//
1127// Implements new-style call filters, and polyfills them into the previous
1128// scheme.
1129//
1130// Call filters:
1131// Derived types should declare a class `Call` with the following members:
1132// - OnClientInitialMetadata - $VALUE_TYPE = ClientMetadata
1133// - OnServerInitialMetadata - $VALUE_TYPE = ServerMetadata
1134// - OnServerToClientMessage - $VALUE_TYPE = Message
1135// - OnClientToServerMessage - $VALUE_TYPE = Message
1136// - OnServerTrailingMetadata - $VALUE_TYPE = ServerMetadata
1137// - OnFinalize - special, see below
1138// These members define an interception point for a particular event in
1139// the call lifecycle.
1140// The type of these members matters, and is selectable by the class
1141// author. For $INTERCEPTOR_NAME in the above list:
1142// - static inline const NoInterceptor $INTERCEPTOR_NAME:
1143// defines that this filter does not intercept this event.
1144// there is zero runtime cost added to handling that event by this filter.
1145// - void $INTERCEPTOR_NAME($VALUE_TYPE&):
1146// the filter intercepts this event, and can modify the value.
1147// it never fails.
1148// - absl::Status $INTERCEPTOR_NAME($VALUE_TYPE&):
1149// the filter intercepts this event, and can modify the value.
1150// it can fail, in which case the call will be aborted.
1151// - ServerMetadataHandle $INTERCEPTOR_NAME($VALUE_TYPE&)
1152// the filter intercepts this event, and can modify the value.
1153// the filter can return nullptr for success, or a metadata handle for
1154// failure (in which case the call will be aborted).
1155// useful for cases where the exact metadata returned needs to be customized.
1156// - void $INTERCEPTOR_NAME($VALUE_TYPE&, Derived*):
1157// the filter intercepts this event, and can modify the value.
1158// it can access the channel via the second argument.
1159// it never fails.
1160// - absl::Status $INTERCEPTOR_NAME($VALUE_TYPE&, Derived*):
1161// the filter intercepts this event, and can modify the value.
1162// it can access the channel via the second argument.
1163// it can fail, in which case the call will be aborted.
1164// - ServerMetadataHandle $INTERCEPTOR_NAME($VALUE_TYPE&, Derived*)
1165// the filter intercepts this event, and can modify the value.
1166// it can access the channel via the second argument.
1167// the filter can return nullptr for success, or a metadata handle for
1168// failure (in which case the call will be aborted).
1169// useful for cases where the exact metadata returned needs to be customized.
1170// It's also acceptable to return a promise that resolves to the
1171// relevant return type listed above.
1172// Finally, OnFinalize can be added to intecept call finalization.
1173// It must have one of the signatures:
1174// - static inline const NoInterceptor OnFinalize:
1175// the filter does not intercept call finalization.
1176// - void OnFinalize(const grpc_call_final_info*):
1177// the filter intercepts call finalization.
1178class ImplementChannelFilterTag {};
1179template <typename Derived>
1180class ImplementChannelFilter : public ChannelFilter,
1181 public ImplementChannelFilterTag {
1182 public:
1183 // Polyfill for the original promise scheme.
1184 // Allows writing v3 filters that work with v2 stacks.
1185 // (and consequently also v1 stacks since we can polyfill back to that too).
1186 ArenaPromise<ServerMetadataHandle> MakeCallPromise(
1187 CallArgs call_args, NextPromiseFactory next_promise_factory) final {
1188 auto* call = promise_filter_detail::MakeFilterCall<Derived>(
1189 static_cast<Derived*>(this));
1190 promise_filter_detail::InterceptClientToServerMessage<
1191 decltype(&Derived::Call::OnClientToServerMessage),
1192 decltype(&Derived::Call::OnClientToServerHalfClose),
1193 Derived>::Run(call, call_args);
1194 promise_filter_detail::InterceptServerInitialMetadata<
1195 Derived,
1196 decltype(&Derived::Call::OnServerInitialMetadata)>::Run(call,
1197 call_args);
1198 promise_filter_detail::InterceptServerToClientMessage<
1199 Derived,
1200 decltype(&Derived::Call::OnServerToClientMessage)>::Run(call,
1201 call_args);
1202 promise_filter_detail::
1203 InterceptFinalize<Derived, decltype(&Derived::Call::OnFinalize)>::Run(
1204 static_cast<Derived*>(this),
1205 static_cast<typename Derived::Call*>(&call->call));
1206 return promise_filter_detail::MapResult(
1207 &Derived::Call::OnServerTrailingMetadata,
1208 promise_filter_detail::RaceAsyncCompletion<
1209 promise_filter_detail::CallHasAsyncErrorInterceptor<
1210 Derived>::value>::
1211 Run(promise_filter_detail::RunCall(
1212 &Derived::Call::OnClientInitialMetadata,
1213 std::move(call_args), std::move(next_promise_factory),
1214 call),
1215 &call->error_latch),
1216 call);
1217 }
