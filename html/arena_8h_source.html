52// Tracks all registered arena context types (these should only be registered
53// via ArenaContextTraits at static initialization time).
54class BaseArenaContextTraits {
55 public:
56 // Count of number of contexts that have been allocated.
57 static uint16_t NumContexts() {
58 return static_cast<uint16_t>(RegisteredTraits().size());
59 }
60
61 // Number of bytes required to store the context pointers on an arena.
62 static size_t ContextSize() { return NumContexts() * sizeof(void*); }
63
64 // Call the registered destruction function for a context.
65 static void Destroy(uint16_t id, void* ptr) {
66 if (ptr == nullptr) return;
67 RegisteredTraits()[id](ptr);
68 }
69
70 protected:
71 // Allocate a new context id and register the destruction function.
72 static uint16_t MakeId(void (*destroy)(void* ptr)) {
73 auto& traits = RegisteredTraits();
74 const uint16_t id = static_cast<uint16_t>(traits.size());
75 traits.push_back(destroy);
76 return id;
77 }
78
79 private:
80 static std::vector<void (*)(void*)>& RegisteredTraits() {
81 static NoDestruct<std::vector<void (*)(void*)>> registered_traits;
82 return *registered_traits;
83 }
84};

142class Arena final : public RefCounted<Arena, NonPolymorphicRefCount,
143 arena_detail::UnrefDestroy> {
144 public:
145 // Create an arena, with \a initial_size bytes in the first allocated buffer.
146 static RefCountedPtr<Arena> Create(size_t initial_size,
147 RefCountedPtr<ArenaFactory> arena_factory);
148
149 // Destroy all `ManagedNew` allocated objects.
150 // Allows safe destruction of these objects even if they need context held by
151 // the arena.
152 // Idempotent.
153 // TODO(ctiller): eliminate ManagedNew.
154 void DestroyManagedNewObjects();
155
156 // Return the total amount of memory allocated by this arena.
157 size_t TotalUsedBytes() const {
158 return total_used_.load(std::memory_order_relaxed);
159 }
160
161 // Allocate \a size bytes from the arena.
162 void* Alloc(size_t size) {
163 size = GPR_ROUND_UP_TO_ALIGNMENT_SIZE(size);
164 size_t begin = total_used_.fetch_add(size, std::memory_order_relaxed);
165 if (begin + size <= initial_zone_size_) {
166 return reinterpret_cast<char*>(this) + begin;
167 } else {
168 return AllocZone(size);
169 }
170 }
171
172 // Allocates T from the arena.
173 // The caller is responsible for calling p->~T(), but should NOT delete.
174 // TODO(roth): We currently assume that all callers need alignment of 16
175 // bytes, which may be wrong in some cases. When we have time, we should
176 // change this to instead use the alignment of the type being allocated by
177 // this method.
178 template <typename T, typename... Args>
179 T* New(Args&&... args) {
180 T* t = static_cast<T*>(Alloc(sizeof(T)));
181 new (t) T(std::forward<Args>(args)...);
182 return t;
183 }
184
185 // Like New, but has the arena call p->~T() at arena destruction time.
186 // The caller should NOT delete.
187 template <typename T, typename... Args>
188 T* ManagedNew(Args&&... args) {
189 auto* p = New<ManagedNewImpl<T>>(std::forward<Args>(args)...);
190 p->Link(&managed_new_head_);
191 return &p->t;
192 }
193
194 template <typename T, typename... Args>
195 absl::enable_if_t<std::is_same<typename T::RefCountedUnrefBehaviorType,
196 UnrefCallDtor>::value,
197 RefCountedPtr<T>>
198 MakeRefCounted(Args&&... args) {
199 return RefCountedPtr<T>(New<T>(std::forward<Args>(args)...));
200 }

242 template <typename T>
243 using PoolPtr =
244 std::unique_ptr<T, typename arena_detail::IfArray<
245 T, PooledDeleter, ArrayPooledDeleter>::Result>;
246
247 // Make a unique_ptr to T that is allocated from the arena.
248 // When the pointer is released, the memory may be reused for other
249 // MakePooled(.*) calls.
250 // CAUTION: The amount of memory allocated is rounded up to the nearest
251 // value in Arena::PoolSizes, and so this may pessimize total
252 // arena size.
253 template <typename T, typename... Args>
254 static PoolPtr<T> MakePooled(Args&&... args) {
255 return PoolPtr<T>(new T(std::forward<Args>(args)...), PooledDeleter());
256 }
257
258 template <typename T>
259 static PoolPtr<T> MakePooledForOverwrite() {
260 return PoolPtr<T>(new T, PooledDeleter());
261 }
262
263 // Make a unique_ptr to an array of T that is allocated from the arena.
264 // When the pointer is released, the memory may be reused for other
265 // MakePooled(.*) calls.
266 // One can use MakePooledArray<char> to allocate a buffer of bytes.
267 // CAUTION: The amount of memory allocated is rounded up to the nearest
268 // value in Arena::PoolSizes, and so this may pessimize total
269 // arena size.
270 template <typename T>
271 PoolPtr<T[]> MakePooledArray(size_t n) {
272 return PoolPtr<T[]>(new T[n], ArrayPooledDeleter());
273 }
274
275 // Like MakePooled, but with manual memory management.
276 // The caller is responsible for calling DeletePooled() on the returned
277 // pointer, and expected to call it with the same type T as was passed to this
278 // function (else the free list returned to the arena will be corrupted).
279 template <typename T, typename... Args>
280 T* NewPooled(Args&&... args) {
281 return new T(std::forward<Args>(args)...);
282 }

289 // Context accessors
290 // Prefer to use the free-standing `GetContext<>` and `SetContext<>` functions
291 // for modern promise-based code -- however legacy filter stack based code
292 // often needs to access these directly.
293 template <typename T>
294 GPR_ATTRIBUTE_ALWAYS_INLINE_FUNCTION T* GetContext() {
295 return static_cast<T*>(
296 contexts()[arena_detail::ArenaContextTraits<T>::id()]);
297 }
298
299 template <typename T>
300 void SetContext(T* context) {
301 void*& slot = contexts()[arena_detail::ArenaContextTraits<T>::id()];
302 if (slot != nullptr) {
303 ArenaContextType<T>::Destroy(static_cast<T*>(slot));
304 }
305 slot = context;
306 DCHECK_EQ(GetContext<T>(), context);
307 }

336 // Initialize an arena.
337 // Parameters:
338 // initial_size: The initial size of the whole arena in bytes. These bytes
339 // are contained within 'zone 0'. If the arena user ends up requiring more
340 // memory than the arena contains in zone 0, subsequent zones are allocated
341 // on demand and maintained in a tail-linked list.
342 //
343 // initial_alloc: Optionally, construct the arena as though a call to
344 // Alloc() had already been made for initial_alloc bytes. This provides a
345 // quick optimization (avoiding an atomic fetch-add) for the common case
346 // where we wish to create an arena and then perform an immediate
347 // allocation.
348 explicit Arena(size_t initial_size,
349 RefCountedPtr<ArenaFactory> arena_factory);
350
351 ~Arena();
352
353 void* AllocZone(size_t size);
354 void Destroy() const;
355 GPR_ATTRIBUTE_ALWAYS_INLINE_FUNCTION void** contexts() {
356 return reinterpret_cast<void**>(this + 1);
357 }
358
359 // Keep track of the total used size. We use this in our call sizing
360 // hysteresis.
361 const size_t initial_zone_size_;
362 std::atomic<size_t> total_used_;
363 std::atomic<size_t> total_allocated_{initial_zone_size_};
364 // If the initial arena allocation wasn't enough, we allocate additional zones
365 // in a reverse linked list. Each additional zone consists of (1) a pointer to
366 // the zone added before this zone (null if this is the first additional zone)
367 // and (2) the allocated memory. The arena itself maintains a pointer to the
368 // last zone; the zone list is reverse-walked during arena destruction only.
369 std::atomic<Zone*> last_zone_{nullptr};
370 std::atomic<ManagedNewObject*> managed_new_head_{nullptr};
371 RefCountedPtr<ArenaFactory> arena_factory_;
372};

374// Arena backed single-producer-single-consumer queue
375// Based on implementation from
376// https://www.1024cores.net/home/lock-free-algorithms/queues/unbounded-spsc-queue
377template <typename T, bool kOptimizeAlignment = true>
378class ArenaSpsc {
379 public:
380 explicit ArenaSpsc(Arena* arena) : arena_(arena) {}
381
382 ~ArenaSpsc() {
383 while (Pop().has_value()) {
384 }
385 }
386
387 ArenaSpsc(const ArenaSpsc&) = delete;
388 ArenaSpsc& operator=(const ArenaSpsc&) = delete;
389
390 // Push `value` onto the queue; at most one thread can be calling this at a
391 // time.
392 void Push(T value) {
393 Node* n = AllocNode();
394 Construct(&n->value, std::move(value));
395 n->next.store(nullptr, std::memory_order_relaxed);
396 head_->next.store(n, std::memory_order_release);
397 head_ = n;
398 }
399
400 // Pop a value from the queue; at most one thread can be calling this at a
401 // time. If the queue was empty when called, returns nullopt.
402 std::optional<T> Pop() {
403 Node* n = tail_.load(std::memory_order_relaxed);
404 Node* next = n->next.load(std::memory_order_acquire);
405 if (next == nullptr) return std::nullopt;
406 T result = std::move(next->value);
407 Destruct(&next->value);
408 tail_.store(next, std::memory_order_release);
409 return result;
410 }
411
412 private:
413 struct Node {
414 Node() {}
415 ~Node() {}
416 explicit Node(std::nullptr_t) : next{nullptr} {}
417 std::atomic<Node*> next;
418 union {
419 T value;
420 };
421 };
422
423 Node* AllocNode() {
424 if (first_ != tail_copy_) {
425 Node* n = first_;
426 first_ = first_->next.load(std::memory_order_relaxed);
427 return n;
428 }
429 tail_copy_ = tail_.load(std::memory_order_acquire);
430 if (first_ != tail_copy_) {
431 Node* n = first_;
432 first_ = first_->next.load(std::memory_order_relaxed);
433 return n;
434 }
435 return arena_->New<Node>();
436 }
437
438 Arena* const arena_;
439 Node first_node_{nullptr};
440 // Accessed mainly by consumer, infrequently by producer
441 std::atomic<Node*> tail_{&first_node_};
442 // Ensure alignment on next cacheline to deliminate producer and consumer
443 // Head of queue
444 alignas(kOptimizeAlignment ? GPR_CACHELINE_SIZE
445 : alignof(Node*)) Node* head_{&first_node_};
446 // Last unused node
447 Node* first_{&first_node_};
448 // Helper, points somewhere between first and tail
449 Node* tail_copy_{&first_node_};
450};
