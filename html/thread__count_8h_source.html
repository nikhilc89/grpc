34// Tracks counts across some fixed number of shards.
35// It is intended for fast increment/decrement operations, but a slower overall
36// count operation.
37class BusyThreadCount {
38 public:
39 // Increments a per-shard counter on construction, decrements on destruction.
40 class AutoThreadCounter {
41 public:
42 AutoThreadCounter(BusyThreadCount* counter, size_t idx)
43 : counter_(counter), idx_(idx) {
44 counter_->Increment(idx_);
45 }
46 ~AutoThreadCounter() {
47 if (counter_ != nullptr) counter_->Decrement(idx_);
48 }
49 // not copyable
50 AutoThreadCounter(const AutoThreadCounter&) = delete;
51 AutoThreadCounter& operator=(const AutoThreadCounter&) = delete;
52 // moveable
53 AutoThreadCounter(AutoThreadCounter&& other) noexcept {
54 counter_ = std::exchange(other.counter_, nullptr);
55 idx_ = other.idx_;
56 }
57 AutoThreadCounter& operator=(AutoThreadCounter&& other) noexcept {
58 counter_ = std::exchange(other.counter_, nullptr);
59 idx_ = other.idx_;
60 return *this;
61 }
62
63 private:
64 BusyThreadCount* counter_;
65 size_t idx_;
66 };
67
68 BusyThreadCount() : shards_(grpc_core::Clamp(gpr_cpu_num_cores(), 2u, 64u)) {}
69 AutoThreadCounter MakeAutoThreadCounter(size_t idx) {
70 return AutoThreadCounter(this, idx);
71 };
72 void Increment(size_t idx) {
73 shards_[idx].busy_count.fetch_add(1, std::memory_order_relaxed);
74 }
75 void Decrement(size_t idx) {
76 shards_[idx].busy_count.fetch_sub(1, std::memory_order_relaxed);
77 }
78 size_t count() {
79 return std::accumulate(
80 shards_.begin(), shards_.end(), 0, [](size_t running, ShardedData& d) {
81 return running + d.busy_count.load(std::memory_order_relaxed);
82 });
83 }
84 // Returns some valid index into the per-shard data, which is rotated on every
85 // call to distribute load and reduce contention.
86 size_t NextIndex() { return next_idx_.fetch_add(1) % shards_.size(); }
87
88 private:
89 // We want to ensure that this data structure lands on different cachelines
90 // per cpu.
91 struct ShardedData {
92 std::atomic<size_t> busy_count{0};
93 } GPR_ALIGN_STRUCT(GPR_CACHELINE_SIZE);
94
95 std::vector<ShardedData> shards_;
96 std::atomic<size_t> next_idx_{0};
97};

99// Tracks the number of living threads. It is intended for a fast count
100// operation, with relatively slower increment/decrement operations.
101class LivingThreadCount {
102 public:
103 // Increments the global counter on construction, decrements on destruction.
104 class AutoThreadCounter {
105 public:
106 explicit AutoThreadCounter(LivingThreadCount* counter) : counter_(counter) {
107 counter_->Increment();
108 }
109 ~AutoThreadCounter() {
110 if (counter_ != nullptr) counter_->Decrement();
111 }
112 // not copyable
113 AutoThreadCounter(const AutoThreadCounter&) = delete;
114 AutoThreadCounter& operator=(const AutoThreadCounter&) = delete;
115 // moveable
116 AutoThreadCounter(AutoThreadCounter&& other) noexcept {
117 counter_ = std::exchange(other.counter_, nullptr);
118 }
119 AutoThreadCounter& operator=(AutoThreadCounter&& other) noexcept {
120 counter_ = std::exchange(other.counter_, nullptr);
121 return *this;
122 }
123
124 private:
125 LivingThreadCount* counter_;
126 };
127
128 AutoThreadCounter MakeAutoThreadCounter() { return AutoThreadCounter(this); };
129 void Increment() ABSL_LOCKS_EXCLUDED(mu_) {
130 grpc_core::MutexLock lock(&mu_);
131 ++living_count_;
132 cv_.SignalAll();
133 }
134 void Decrement() ABSL_LOCKS_EXCLUDED(mu_) {
135 grpc_core::MutexLock lock(&mu_);
136 --living_count_;
137 cv_.SignalAll();
138 }
139 // Blocks the calling thread until the desired number of threads is reached.
140 // If the thread count does not change for some given `stuck_timeout`
141 // duration, this method returns error. If the thread count does change, the
142 // timeout clock is reset.
143 absl::Status BlockUntilThreadCount(size_t desired_threads, const char* why,
144 grpc_core::Duration stuck_timeout)
145 ABSL_LOCKS_EXCLUDED(mu_);
146 size_t count() ABSL_LOCKS_EXCLUDED(mu_) {
147 grpc_core::MutexLock lock(&mu_);
148 return CountLocked();
149 }
150
151 private:
152 size_t CountLocked() ABSL_EXCLUSIVE_LOCKS_REQUIRED(mu_) {
153 return living_count_;
154 }
155 size_t WaitForCountChange(size_t desired_threads,
156 grpc_core::Duration timeout);
157
158 grpc_core::Mutex mu_;
159 grpc_core::CondVar cv_ ABSL_GUARDED_BY(mu_);
160 size_t living_count_ ABSL_GUARDED_BY(mu_) = 0;
161};
