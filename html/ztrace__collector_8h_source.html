69template <typename T>
70void AppendResults(const Collection<T>& data, Json::Array& results) {
71 for (const auto& value : data) {
72 Json::Object object;
73 object["timestamp"] =
74 Json::FromString(gpr_format_timespec(gpr_convert_clock_type(
75 gpr_cycle_counter_to_time(value.first), GPR_CLOCK_REALTIME)));
76 value.second.RenderJson(object);
77 results.emplace_back(Json::FromObject(std::move(object)));
78 }

102// Generic collector infrastructure for ztrace queries.
103// Abstracts away most of the ztrace requirements in an efficient manner,
104// allowing system authors to concentrate on emitting useful data.
105// If no trace is performed, overhead is one pointer and one relaxed atomic read
106// per trace event.
107//
108// Two kinds of objects are required:
109// 1. A `Config`
110// - This type should be constructible with a std::map<std::string,
111// std::string>
112// and provides overall query configuration - the map can be used to pull
113// predicates from the calling system.
114// - Needs a `bool Finishes(T)` method for each Data type (see 2).
115// This allows the config to terminate a query in the event of reaching
116// some configured predicate.
117// 2. N `Data` types
118// - One for each kind of data captured in the trace
119// - Allows avoiding e.g. variant<> data types; these are inefficient
120// in this context because they force every recorded entry to use the
121// same number of bytes whilst pending.
122template <typename Config, typename... Data>
123class ZTraceCollector {
124 public:
125 template <typename X>
126 void Append(X producer_or_value) {
127 if (!impl_.is_set()) return;
128 if constexpr (ztrace_collector_detail::kIsElement<X, Data...>) {
129 AppendValue(std::move(producer_or_value));
130 } else {
131 AppendValue(producer_or_value());
132 }
133 }

143 struct Instance : public RefCounted<Instance> {
144 Instance(std::map<std::string, std::string> args,
145 std::shared_ptr<grpc_event_engine::experimental::EventEngine>
146 event_engine,
147 absl::AnyInvocable<void(Json)> done)
148 : memory_cap_(IntFromArgs(args, "memory_cap").value_or(1024 * 1024)),
149 config(args),
150 event_engine(std::move(event_engine)),
151 done(std::move(done)) {}
152 using Collections = std::tuple<Collection<Data>...>;
153 struct RemoveMostRecentState {
154 void (*enact)(Instance*) = nullptr;
155 gpr_cycle_counter most_recent;
156 };
157 template <typename T>
158 void Append(std::pair<gpr_cycle_counter, T> value) {
159 memory_used_ += value.second.MemoryUsage();
160 while (memory_used_ > memory_cap_) RemoveMostRecent();
161 std::get<Collection<T> >(data).push_back(std::move(value));
162 }
163 void RemoveMostRecent() {
164 RemoveMostRecentState state;
165 (UpdateRemoveMostRecentState<Data>(&state), ...);
166 CHECK(state.enact != nullptr);
167 state.enact(this);
168 }
169 template <typename T>
170 void UpdateRemoveMostRecentState(RemoveMostRecentState* state) {
171 auto& collection = std::get<Collection<T> >(data);
172 if (collection.empty()) return;
173 if (state->enact == nullptr ||
174 collection.front().first > state->most_recent) {
175 state->enact = +[](Instance* instance) {
176 auto& collection = std::get<Collection<T> >(instance->data);
177 const size_t ent_usage = collection.front().second.MemoryUsage();
178 CHECK_GE(instance->memory_used_, ent_usage);
179 instance->memory_used_ -= ent_usage;
180 collection.pop_front();
181 };
182 state->most_recent = collection.front().first;
183 }
184 }
185 void Finish(absl::Status status) {
186 event_engine->Run([data = std::move(data), done = std::move(done),
187 status = std::move(status),
188 memory_used = memory_used_]() mutable {
189 Json::Array entries;
190 (ztrace_collector_detail::AppendResults(
191 std::get<Collection<Data> >(data), entries),
192 ...);
193 Json::Object result;
194 result["entries"] = Json::FromArray(entries);
195 result["status"] = Json::FromString(status.ToString());
196 result["memory_used"] =
197 Json::FromNumber(static_cast<uint64_t>(memory_used));
198 done(Json::FromObject(std::move(result)));
199 });
200 }
201 size_t memory_used_ = 0;
202 size_t memory_cap_ = 0;
203 Config config;
204 const Timestamp start_time = Timestamp::Now();
205 std::shared_ptr<grpc_event_engine::experimental::EventEngine> event_engine;
206 grpc_event_engine::experimental::EventEngine::TaskHandle task_handle{
207 grpc_event_engine::experimental::EventEngine::TaskHandle::kInvalid};
208 Collections data;
209 absl::AnyInvocable<void(Json)> done;
210 };

216 public:
217 explicit ZTraceImpl(RefCountedPtr<Impl> impl) : impl_(std::move(impl)) {}
218
219 void Run(Timestamp deadline, std::map<std::string, std::string> args,
220 std::shared_ptr<grpc_event_engine::experimental::EventEngine>
221 event_engine,
222 absl::AnyInvocable<void(Json)> callback) override {
223 auto instance = MakeRefCounted<Instance>(std::move(args), event_engine,
224 std::move(callback));
225 auto impl = std::move(impl_);
226 RefCountedPtr<Instance> oldest_instance;
227 MutexLock lock(&impl->mu);
228 if (impl->instances.size() > 20) {
229 // Eject oldest running trace
230 Timestamp oldest_time = Timestamp::InfFuture();
231 for (auto& instance : impl->instances) {
232 if (instance->start_time < oldest_time) {
233 oldest_time = instance->start_time;
234 oldest_instance = instance;
235 }
236 }
237 CHECK(oldest_instance != nullptr);
238 impl->instances.erase(oldest_instance);
239 oldest_instance->Finish(
240 absl::ResourceExhaustedError("Too many concurrent ztrace queries"));
241 }
242 instance->task_handle = event_engine->RunAfter(
243 deadline - Timestamp::Now(), [instance, impl]() {
244 bool finish;
245 {
246 MutexLock lock(&impl->mu);
247 finish = impl->instances.erase(instance);
248 }
249 if (finish) instance->Finish(absl::DeadlineExceededError(""));
250 });
251 impl->instances.insert(instance);
252 }
253
254 private:
255 RefCountedPtr<Impl> impl_;
256 };

258 template <typename T>
259 void AppendValue(T&& data) {
260 auto value = std::pair(gpr_get_cycle_counter(), std::forward<T>(data));
261 auto* impl = impl_.Get();
262 {
263 MutexLock lock(&impl->mu);
264 switch (impl->instances.size()) {
265 case 0:
266 return;
267 case 1: {
268 auto& instances = impl->instances;
269 auto& instance = *instances.begin();
270 const bool finishes = instance->config.Finishes(value.second);
271 instance->Append(std::move(value));
272 if (finishes) {
273 instance->Finish(absl::OkStatus());
274 instances.clear();
275 }
276 } break;
277 default: {
278 std::vector<RefCountedPtr<Instance> > finished;
279 for (auto& instance : impl->instances) {
280 const bool finishes = instance->config.Finishes(value.second);
281 instance->Append(value);
282 if (finishes) {
283 finished.push_back(instance);
284 }
285 }
286 for (const auto& instance : finished) {
287 instance->Finish(absl::OkStatus());
288 impl->instances.erase(instance);
289 }
290 }
291 }
292 }
293 }

295 SingleSetRefCountedPtr<Impl> impl_;
