42// Multi Producer Single Consumer (MPSC) inter-activity communications.
43// MPSC is used to communicate in between two or more Activities or Promise
44// Parties in a thread safe way.
45// The communication consists of one or more MpscSender objects and one
46// MpscReceiver.
47
48// "Center" of the communication pipe.
49// Contains sent but not received messages, and open/close state.
50template <typename T>
51class Center : public RefCounted<Center<T>> {
52 public:
53 // Construct the center with a maximum queue size.
54 explicit Center(size_t max_queued) : max_queued_(max_queued) {}
55
56 static constexpr const uint64_t kClosedBatch =
57 std::numeric_limits<uint64_t>::max();
58
59 // Poll for new items.
60 // - Returns true if new items were obtained, in which case they are contained
61 // in dest in the order they were added. Wakes up all pending senders since
62 // there will now be space to send.
63 // - If receives have been closed, returns false.
64 // - If no new items are available, returns
65 // Pending and sets up a waker to be awoken when more items are available.
66 // TODO(ctiller): consider the problem of thundering herds here. There may be
67 // more senders than there are queue spots, and so the strategy of waking up
68 // all senders is ill-advised.
69 // That said, some senders may have been cancelled by the time we wake them,
70 // and so waking a subset could cause starvation.
71 Poll<bool> PollReceiveBatch(std::vector<T>& dest) {
72 ReleasableMutexLock lock(&mu_);
73 GRPC_TRACE_LOG(promise_primitives, INFO)
74 << "MPSC::PollReceiveBatch: "
75 << GRPC_DUMP_ARGS(this, batch_, queue_.size());
76 if (queue_.empty()) {
77 if (batch_ == kClosedBatch) return false;
78 receive_waker_ = GetContext<Activity>()->MakeNonOwningWaker();
79 return Pending{};
80 }
81 dest.swap(queue_);
82 queue_.clear();
83 if (batch_ != kClosedBatch) ++batch_;
84 auto wakeups = send_wakers_.TakeWakeupSet();
85 lock.Release();
86 wakeups.Wakeup();
87 return true;
88 }
89
90 // Return value:
91 // - if the pipe is closed, returns kClosedBatch
92 // - if await_receipt is false, returns the batch number the item was sent
93 // in.
94 // - if await_receipt is true, returns the first sending batch number that
95 // guarantees the item has been received.
96 template <bool kAwaitReceipt>
97 uint64_t Send(T t) {
98 ReleasableMutexLock lock(&mu_);
99 if (batch_ == kClosedBatch) return kClosedBatch;
100 queue_.push_back(std::move(t));
101 auto receive_waker = std::move(receive_waker_);
102 const uint64_t batch =
103 (!kAwaitReceipt && queue_.size() <= max_queued_) ? batch_ : batch_ + 1;
104 lock.Release();
105 receive_waker.Wakeup();
106 return batch;
107 }
108
109 // Poll until a particular batch number is received.
110 Poll<Empty> PollReceiveBatch(uint64_t batch) {
111 ReleasableMutexLock lock(&mu_);
112 GRPC_TRACE_LOG(promise_primitives, INFO)
113 << "MPSC::PollReceiveBatch: " << GRPC_DUMP_ARGS(this, batch_, batch);
114 if (batch_ >= batch) return Empty{};
115 send_wakers_.AddPending(GetContext<Activity>()->MakeNonOwningWaker());
116 return Pending{};
117 }
118
119 // Mark that the receiver is closed.
120 void ReceiverClosed(bool wake_receiver) {
121 ReleasableMutexLock lock(&mu_);
122 GRPC_TRACE_LOG(promise_primitives, INFO)
123 << "MPSC::ReceiverClosed: " << GRPC_DUMP_ARGS(this, batch_);
124 if (batch_ == kClosedBatch) return;
125 batch_ = kClosedBatch;
126 auto wakeups = send_wakers_.TakeWakeupSet();
127 auto receive_waker = std::move(receive_waker_);
128 lock.Release();
129 if (wake_receiver) receive_waker.Wakeup();
130 wakeups.Wakeup();
131 }
132
133 private:
134 Mutex mu_;
135 const size_t max_queued_;
136 std::vector<T> queue_ ABSL_GUARDED_BY(mu_);
137 // Every time we give queue_ to the receiver, we increment batch_.
138 // When the receiver is closed we set batch_ to kClosedBatch.
139 uint64_t batch_ ABSL_GUARDED_BY(mu_) = 1;
140 Waker receive_waker_ ABSL_GUARDED_BY(mu_);
141 WaitSet send_wakers_ ABSL_GUARDED_BY(mu_);
142};

149// Send half of an mpsc pipe.
150template <typename T>
151class MpscSender {
152 public:
153 MpscSender() = default;
154 MpscSender(const MpscSender&) = default;
155 MpscSender& operator=(const MpscSender&) = default;
156 MpscSender(MpscSender&&) noexcept = default;
157 MpscSender& operator=(MpscSender&&) noexcept = default;
158
159 // Input: Input is the object that you want to send. The promise that is
160 // returned by Send will take ownership of the object.
161 // Return: Returns a promise that will send one item.
162 // This promise can either return
163 // 1. Pending{} if the sending is still pending
164 // 2. Resolves to true if sending is successful
165 // 3. Resolves to false if the receiver was closed and the value
166 // will never be successfully sent.
167 // The promise returned is thread safe. We can use multiple send calls
168 // in parallel to generate multiple such send promises and these promises can
169 // be run in parallel in a thread safe way.
170 auto Send(T t) { return SendGeneric<false>(std::move(t)); }
171
172 // Similar to send, but the promise returned by SendAcked will not resolve
173 // until the item has been received by the receiver.
174 auto SendAcked(T t) { return SendGeneric<true>(std::move(t)); }
175
176 StatusFlag UnbufferedImmediateSend(T t) {
177 return StatusFlag(center_->template Send<false>(std::move(t)) !=
178 mpscpipe_detail::Center<T>::kClosedBatch);
179 }
180
181 private:
182 template <bool kAwaitReceipt>
183 auto SendGeneric(T t) {
184 return [center = center_, t = std::move(t),
185 batch = uint64_t(0)]() mutable -> Poll<StatusFlag> {
186 if (center == nullptr) return Failure{};
187 if (batch == 0) {
188 batch = center->template Send<kAwaitReceipt>(std::move(t));
189 DCHECK_NE(batch, 0u);
190 if (batch == mpscpipe_detail::Center<T>::kClosedBatch) return Failure{};
191 }
192 auto p = center->PollReceiveBatch(batch);
193 if (p.pending()) return Pending{};
194 return Success{};
195 };
196 }
197
198 friend class MpscReceiver<T>;
199 explicit MpscSender(RefCountedPtr<mpscpipe_detail::Center<T>> center)
200 : center_(std::move(center)) {}
201 RefCountedPtr<mpscpipe_detail::Center<T>> center_;
202};

204// Receive half of an mpsc pipe.
205template <typename T>
206class MpscReceiver {
207 public:
208 // max_buffer_hint is the maximum number of elements we'd like to buffer.
209 // We half this before passing to Center so that the number there is the
210 // maximum number of elements that can be queued in the center of the pipe.
211 // The receiver also holds some of the buffered elements (up to half of them!)
212 // so the total outstanding is equal to max_buffer_hint (unless it's 1 in
213 // which case instantaneosly we may have two elements buffered).
214 explicit MpscReceiver(size_t max_buffer_hint)
215 : center_(MakeRefCounted<mpscpipe_detail::Center<T>>(
216 std::max(static_cast<size_t>(1), max_buffer_hint / 2))) {}
217 ~MpscReceiver() {
218 if (center_ != nullptr) center_->ReceiverClosed(false);
219 }
220 // Marking the receiver closed will make sure it will not receive any
221 // messages. If a sender tries to Send a message to a closed receiver,
222 // sending will fail.
223 void MarkClosed() {
224 if (center_ != nullptr) center_->ReceiverClosed(true);
225 }
226 MpscReceiver(const MpscReceiver&) = delete;
227 MpscReceiver& operator=(const MpscReceiver&) = delete;
228 // Only movable until it's first polled, and so we don't need to contend with
229 // a non-empty buffer during a legal move!
230 MpscReceiver(MpscReceiver&& other) noexcept
231 : center_(std::move(other.center_)) {
232 DCHECK(other.buffer_.empty());
233 }
234 MpscReceiver& operator=(MpscReceiver&& other) noexcept {
235 DCHECK(other.buffer_.empty());
236 center_ = std::move(other.center_);
237 return *this;
238 }
239
240 // Construct a new sender for this receiver. One receiver can have multiple
241 // senders.
242 MpscSender<T> MakeSender() { return MpscSender<T>(center_); }
243
244 // Returns a promise that will resolve to ValueOrFailure<T>.
245 // If receiving is closed, the promise will resolve to failure.
246 // Otherwise, the promise resolves to the next item and removes
247 // said item from the queue.
248 auto Next() {
249 return [this]() -> Poll<ValueOrFailure<T>> {
250 if (buffer_it_ != buffer_.end()) {
251 return Poll<ValueOrFailure<T>>(std::move(*buffer_it_++));
252 }
253 auto p = center_->PollReceiveBatch(buffer_);
254 if (bool* r = p.value_if_ready()) {
255 if (!*r) return Failure{};
256 buffer_it_ = buffer_.begin();
257 return Poll<ValueOrFailure<T>>(std::move(*buffer_it_++));
258 }
259 return Pending{};
260 };
261 }
262 // Returns a promise that will resolve to ValueOrFailure<std::vector<T>>.
263 // If receiving is closed, the promise will resolve to failure.
264 // Otherwise, the promise returns all the items enqueued till now and removes
265 // said items from the queue.
266 auto NextBatch() {
267 return [this]() -> Poll<ValueOrFailure<std::vector<T>>> {
268 if (buffer_it_ != buffer_.end()) {
269 std::vector<T> tmp_buffer;
270 std::move(buffer_it_, buffer_.end(), std::back_inserter(tmp_buffer));
271 buffer_.clear();
272 buffer_it_ = buffer_.end();
273 return ValueOrFailure<std::vector<T>>(std::move(tmp_buffer));
274 }
275
276 auto p = center_->PollReceiveBatch(buffer_);
277 if (bool* r = p.value_if_ready()) {
278 if (!*r) return Failure{};
279 auto tmp_buffer(std::move(buffer_));
280 buffer_.clear();
281 buffer_it_ = buffer_.end();
282 return ValueOrFailure<std::vector<T>>(std::move(tmp_buffer));
283 }
284 return Pending{};
285 };
286 }
287
288 private:
289 // Received items. We move out of here one by one, but don't resize the
290 // vector. Instead, when we run out of items, we poll the center for more -
291 // which swaps this buffer in for the new receive queue and clears it.
292 // In this way, upon hitting a steady state the queue ought to be allocation
293 // free.
294 std::vector<T> buffer_;
295 typename std::vector<T>::iterator buffer_it_ = buffer_.end();
296 RefCountedPtr<mpscpipe_detail::Center<T>> center_;
297};
