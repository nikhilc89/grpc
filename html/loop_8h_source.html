32// A Loop combinator
33//
34// Input:
35//
36// 1. A Loop combinator takes as input only one promise factory.
37// 2. This input promise factory should have a return type of either
38// a. LoopCtl<T> which is an alias for std::variant<Continue, T>
39// b. Or Poll<LoopCtl<T>>
40//
41// Running of the Loop combinator:
42//
43// 1. The input promise is guranteed to run at least once when the combinator
44// is invoked.
45// 2. The Loop combinators execution will keep running input promise for
46// as long as the input promise returns the Continue() object.
47// 3. The Loop breaks if
48// a. the input promise returns T.
49// b. the input promise returns Pending{}.
50//
51// The execution of multiple iterations of the input
52// promise happen on the same thread.
53//
54// Return:
55//
56// The Loop combinator when executed will return Poll<T>.
57//
58// Example:
59//
60// {
61// std::string execution_order;
62// int i = 0;
63// Poll<int> retval = Loop([&execution_order, &i]() {
64// return [&execution_order, &i]() -> LoopCtl<int> {
65// absl::StrAppend(&execution_order, i);
66// i++;
67// if (i < 5) return Continue();
68// return i;
69// };
70// })();
71// EXPECT_TRUE(retval.ready());
72// EXPECT_EQ(retval.value(), 5);
73// EXPECT_EQ(i, 5);
74// EXPECT_STREQ(execution_order.c_str(), "01234");
75// }
76
77// Special type - signals to loop to take another iteration, instead of
78// finishing
79struct Continue {};
80
81// Result of polling a loop promise - either Continue looping, or return a value
82// T
83template <typename T>
84using LoopCtl = std::variant<Continue, T>;

86namespace promise_detail {

88template <typename T>
89struct LoopTraits;
90
91template <typename T>
92struct LoopTraits<LoopCtl<T>> {
93 using Result = T;
94 GPR_ATTRIBUTE_ALWAYS_INLINE_FUNCTION static LoopCtl<T> ToLoopCtl(
95 LoopCtl<T> value) {
96 return value;
97 }
98};
99
100template <typename T>
101struct LoopTraits<absl::StatusOr<LoopCtl<T>>> {
102 using Result = absl::StatusOr<T>;
103 GPR_ATTRIBUTE_ALWAYS_INLINE_FUNCTION static LoopCtl<Result> ToLoopCtl(
104 absl::StatusOr<LoopCtl<T>> value) {
105 if (!value.ok()) return value.status();
106 auto& inner = *value;
107 if (std::holds_alternative<Continue>(inner)) return Continue{};
108 return std::get<T>(std::move(inner));
109 }
110};
111
112template <>
113struct LoopTraits<absl::StatusOr<LoopCtl<absl::Status>>> {
114 using Result = absl::Status;
115 GPR_ATTRIBUTE_ALWAYS_INLINE_FUNCTION static LoopCtl<Result> ToLoopCtl(
116 absl::StatusOr<LoopCtl<absl::Status>> value) {
117 if (!value.ok()) return value.status();
118 const auto& inner = *value;
119 if (std::holds_alternative<Continue>(inner)) return Continue{};
120 return std::get<absl::Status>(inner);
121 }
122};

126template <typename F>
127class Loop {
128 private:
129 static_assert(promise_detail::kIsRepeatedPromiseFactory<void, F>);
130
131 using Factory = promise_detail::RepeatedPromiseFactory<void, F>;
132 using PromiseType = decltype(std::declval<Factory>().Make());
133 using PromiseResult = typename PromiseType::Result;
134
135 public:
136 using Result = typename promise_detail::LoopTraits<PromiseResult>::Result;
137
138 GPR_ATTRIBUTE_ALWAYS_INLINE_FUNCTION explicit Loop(F f)
139 : factory_(std::move(f)) {}
140 GPR_ATTRIBUTE_ALWAYS_INLINE_FUNCTION ~Loop() {
141 if (started_) Destruct(&promise_);
142 }
143
144 GPR_ATTRIBUTE_ALWAYS_INLINE_FUNCTION Loop(Loop&& loop) noexcept
145 : factory_(std::move(loop.factory_)), started_(loop.started_) {
146 if (started_) Construct(&promise_, std::move(loop.promise_));
147 }
148
149 Loop(const Loop& loop) = delete;
150 Loop& operator=(const Loop& loop) = delete;
151
152 GPR_ATTRIBUTE_ALWAYS_INLINE_FUNCTION Poll<Result> operator()() {
153 GRPC_TRACE_LOG(promise_primitives, INFO)
154 << "loop[" << this << "] begin poll started=" << started_;
155 if (!started_) {
156 started_ = true;
157 Construct(&promise_, factory_.Make());
158 }
159 while (true) {
160 // Poll the inner promise.
161 auto promise_result = promise_();
162 // If it returns a value:
163 if (auto* p = promise_result.value_if_ready()) {
164 // - then if it's Continue, destroy the promise and recreate a new one
165 // from our factory.
166 auto lc =
167 promise_detail::LoopTraits<PromiseResult>::ToLoopCtl(std::move(*p));
168 if (std::holds_alternative<Continue>(lc)) {
169 GRPC_TRACE_LOG(promise_primitives, INFO)
170 << "loop[" << this << "] iteration complete, continue";
171 Destruct(&promise_);
172 Construct(&promise_, factory_.Make());
173 continue;
174 }
175 GRPC_TRACE_LOG(promise_primitives, INFO)
176 << "loop[" << this << "] iteration complete, return";
177 // - otherwise there's our result... return it out.
178 return std::get<Result>(std::move(lc));
179 } else {
180 // Otherwise the inner promise was pending, so we are pending.
181 GRPC_TRACE_LOG(promise_primitives, INFO)
182 << "loop[" << this << "] pending";
183 return Pending();
184 }
185 }
186 }
187
188 private:
189 GPR_NO_UNIQUE_ADDRESS Factory factory_;
190 GPR_NO_UNIQUE_ADDRESS union {
191 GPR_NO_UNIQUE_ADDRESS PromiseType promise_;
192 };
193 bool started_ = false;
194};

197Loop(F) -> Loop<F>;
