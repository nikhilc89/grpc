81// Generic implementation of a transport-bound frame (stream_id is zero always)
82// with a protobuf specified body.
83
84template <FrameType frame_type, typename Body>
85struct ProtoTransportFrame final : public FrameInterface {
86 ProtoTransportFrame() = default;
87 explicit ProtoTransportFrame(Body body) : body(std::move(body)) {}
88 absl::Status Deserialize(const FrameHeader& header,
89 SliceBuffer payload) override {
90 DCHECK_EQ(header.type, frame_type);
91 return ReadTransportProto(header, std::move(payload), body);
92 }
93 FrameHeader MakeHeader() const override {
94 return FrameHeader{frame_type, 0, ProtoPayloadSize(body)};
95 }
96 void SerializePayload(SliceBuffer& payload) const override {
97 WriteProto(body, payload);
98 }
99 std::string ToString() const override {
100 return absl::StrCat(FrameTypeString(frame_type), "{",
101 body.ShortDebugString(), "}");
102 }
103 bool operator==(const ProtoTransportFrame& other) const {
104 return body.ShortDebugString() == other.body.ShortDebugString();
105 }
106 bool IsLegal() const override { return true; }
107
108 Body body;
109};

111// Generic implementation of a stream-bound frame with a protobuf specified
112// body.
113template <FrameType frame_type, typename Body>
114struct ProtoStreamFrame final : public FrameInterface {
115 ProtoStreamFrame() = default;
116 ProtoStreamFrame(Body body, uint32_t stream_id)
117 : body(std::move(body)), stream_id(stream_id) {}
118 absl::Status Deserialize(const FrameHeader& header,
119 SliceBuffer payload) override {
120 DCHECK_EQ(header.type, frame_type);
121 return ReadStreamProto(header, std::move(payload), body, stream_id);
122 }
123 FrameHeader MakeHeader() const override {
124 return FrameHeader{frame_type, stream_id, ProtoPayloadSize(body)};
125 }
126 void SerializePayload(SliceBuffer& payload) const override {
127 DCHECK_NE(stream_id, 0u);
128 WriteProto(body, payload);
129 }
130 std::string ToString() const override {
131 return absl::StrCat(FrameTypeString(frame_type), "{@", stream_id, "; ",
132 body.ShortDebugString(), "}");
133 }
134 bool operator==(const ProtoStreamFrame& other) const {
135 return body.ShortDebugString() == other.body.ShortDebugString() &&
136 stream_id == other.stream_id;
137 }
138 bool IsLegal() const override { return stream_id != 0; }
139
140 Body body;
141 uint32_t stream_id;
142};

144// Generic implementation of an empty stream frame.
145template <FrameType frame_type>
146struct EmptyStreamFrame final : public FrameInterface {
147 EmptyStreamFrame() = default;
148 explicit EmptyStreamFrame(uint32_t stream_id) : stream_id(stream_id) {}
149 absl::Status Deserialize(const FrameHeader& header, SliceBuffer) override {
150 DCHECK_EQ(header.type, frame_type);
151 return ReadEmptyFrame(header, stream_id);
152 }
153 FrameHeader MakeHeader() const override {
154 return FrameHeader{frame_type, stream_id, 0};
155 }
156 void SerializePayload(SliceBuffer&) const override {}
157 std::string ToString() const override { return FrameTypeString(frame_type); }
158 bool operator==(const EmptyStreamFrame& other) const {
159 return stream_id == other.stream_id;
160 }
161 bool IsLegal() const override { return stream_id != 0; }
162
163 uint32_t stream_id;
164};

182struct MessageFrame final : public FrameInterface {
183 MessageFrame() = default;
184 MessageFrame(uint32_t stream_id, MessageHandle message)
185 : stream_id(stream_id), message(std::move(message)) {}
186 absl::Status Deserialize(const FrameHeader& header,
187 SliceBuffer payload) override;
188 FrameHeader MakeHeader() const override;
189 void SerializePayload(SliceBuffer& payload) const override;
190 std::string ToString() const override;
191 bool operator==(const MessageFrame& other) const {
192 return stream_id == other.stream_id &&
193 message->payload()->JoinIntoString() ==
194 other.message->payload()->JoinIntoString();
195 }
196 bool IsLegal() const override { return stream_id != 0; }
197
198 uint32_t stream_id;
199 MessageHandle message;
200};

227inline Frame CopyFrame(const Frame& frame) {
228 return Match(
229 frame, [](const auto& frame) -> Frame { return frame; },
230 [](const MessageFrame& frame) -> Frame {
231 return MessageFrame(frame.stream_id, frame.message->Clone());
232 },
233 [](const MessageChunkFrame& frame) -> Frame {
234 return MessageChunkFrame(frame.stream_id, frame.payload.Copy());
235 });
236}
