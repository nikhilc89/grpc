97 // Start and cancel watch for a resource.
98 //
99 // The XdsClient takes ownership of the watcher, but the caller may
100 // keep a raw pointer to the watcher, which may be used only for
101 // cancellation. (Because the caller does not own the watcher, the
102 // pointer must not be used for any other purpose.)
103 // If the caller is going to start a new watch after cancelling the
104 // old one, it should set delay_unsubscription to true.
105 //
106 // The resource type object must be a global singleton, since the first
107 // time the XdsClient sees a particular resource type object, it will
108 // store the pointer to that object as the authoritative implementation for
109 // its type URLs. The resource type object must outlive the XdsClient object,
110 // and it is illegal to start a subsequent watch for the same type URLs using
111 // a different resource type object.
112 //
113 // Note: Most callers will not use this API directly but rather via a
114 // resource-type-specific wrapper API provided by the relevant
115 // XdsResourceType implementation.
116 void WatchResource(const XdsResourceType* type, absl::string_view name,
117 RefCountedPtr<ResourceWatcherInterface> watcher);
118 void CancelResourceWatch(const XdsResourceType* type,
119 absl::string_view listener_name,
120 ResourceWatcherInterface* watcher,
121 bool delay_unsubscription = false);

266 // Resource status from the view of a xDS client, which tells the
267 // synchronization status between the xDS client and the xDS server.
268 enum ClientResourceStatus {
269 // Client requested this resource but hasn't received any update from
270 // management server. The client will not fail requests, but will queue
271 // them until update arrives or the client times out waiting for the
272 // resource.
273 REQUESTED = 1,
274 // This resource has been requested by the client but has either not been
275 // delivered by the server or was previously delivered by the server and
276 // then subsequently removed from resources provided by the server.
277 DOES_NOT_EXIST,
278 // Client received this resource and replied with ACK.
279 ACKED,
280 // Client received this resource and replied with NACK.
281 NACKED,
282 // Server sent an error for the resource.
283 RECEIVED_ERROR,
284 // Client encountered timeout getting resource from server.
285 TIMEOUT,
286 };
287 static_assert(static_cast<ClientResourceStatus>(envoy_admin_v3_REQUESTED) ==
288 ClientResourceStatus::REQUESTED);
289 static_assert(
290 static_cast<ClientResourceStatus>(envoy_admin_v3_DOES_NOT_EXIST) ==
291 ClientResourceStatus::DOES_NOT_EXIST);
292 static_assert(static_cast<ClientResourceStatus>(envoy_admin_v3_ACKED) ==
293 ClientResourceStatus::ACKED);
294 static_assert(static_cast<ClientResourceStatus>(envoy_admin_v3_NACKED) ==
295 ClientResourceStatus::NACKED);
296 static_assert(
297 static_cast<ClientResourceStatus>(envoy_admin_v3_RECEIVED_ERROR) ==
298 ClientResourceStatus::RECEIVED_ERROR);
299 static_assert(static_cast<ClientResourceStatus>(envoy_admin_v3_TIMEOUT) ==
300 ClientResourceStatus::TIMEOUT);
