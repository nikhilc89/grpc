152// One batch operation
153// Wrapper around promise steps to perform once of the batch operations for the
154// legacy grpc surface api.
155template <typename SetupResult, grpc_op_type kOp>
156class OpHandlerImpl {
157 public:
158 using PromiseFactory = promise_detail::OncePromiseFactory<void, SetupResult>;
159 using Promise = typename PromiseFactory::Promise;
160 static_assert(!std::is_same<Promise, void>::value,
161 "PromiseFactory must return a promise");
162
163 OpHandlerImpl() : state_(State::kDismissed) {}
164 explicit OpHandlerImpl(SetupResult result) : state_(State::kPromiseFactory) {
165 Construct(&promise_factory_, std::move(result));
166 }
167
168 ~OpHandlerImpl() {
169 switch (state_) {
170 case State::kDismissed:
171 break;
172 case State::kPromiseFactory:
173 Destruct(&promise_factory_);
174 break;
175 case State::kPromise:
176 Destruct(&promise_);
177 break;
178 }
179 }
180
181 OpHandlerImpl(const OpHandlerImpl&) = delete;
182 OpHandlerImpl& operator=(const OpHandlerImpl&) = delete;
183 OpHandlerImpl(OpHandlerImpl&& other) noexcept : state_(other.state_) {
184 switch (state_) {
185 case State::kDismissed:
186 break;
187 case State::kPromiseFactory:
188 Construct(&promise_factory_, std::move(other.promise_factory_));
189 break;
190 case State::kPromise:
191 Construct(&promise_, std::move(other.promise_));
192 break;
193 }
194 }
195 OpHandlerImpl& operator=(OpHandlerImpl&& other) noexcept = delete;
196
197 Poll<StatusFlag> operator()() {
198 switch (state_) {
199 case State::kDismissed:
200 GRPC_TRACE_LOG(call, INFO)
201 << Activity::current()->DebugTag() << "Dismissed " << OpName();
202 return Success{};
203 case State::kPromiseFactory: {
204 GRPC_TRACE_LOG(call, INFO)
205 << Activity::current()->DebugTag() << "Construct " << OpName();
206 auto promise = promise_factory_.Make();
207 Destruct(&promise_factory_);
208 Construct(&promise_, std::move(promise));
209 state_ = State::kPromise;
210 }
211 [[fallthrough]];
212 case State::kPromise: {
213 GRPC_TRACE_LOG(call, INFO)
214 << Activity::current()->DebugTag() << "BeginPoll " << OpName();
215 auto r = poll_cast<StatusFlag>(promise_());
216 GRPC_TRACE_LOG(call, INFO)
217 << Activity::current()->DebugTag() << "EndPoll " << OpName()
218 << " --> "
219 << (r.pending() ? "PENDING" : (r.value().ok() ? "OK" : "FAILURE"));
220 return r;
221 }
222 }
223 GPR_UNREACHABLE_CODE(return Pending{});
224 }
225
226 private:
227 enum class State {
228 kDismissed,
229 kPromiseFactory,
230 kPromise,
231 };
232
233 static const char* OpName() { return GrpcOpTypeName(kOp); }
234
235 // gcc-12 has problems with this being a variant
236 GPR_NO_UNIQUE_ADDRESS State state_;
237 union {
238 PromiseFactory promise_factory_;
239 Promise promise_;
240 };
241};

248class BatchOpIndex {
249 public:
250 BatchOpIndex(const grpc_op* ops, size_t nops) : ops_(ops) {
251 for (size_t i = 0; i < nops; i++) {
252 idxs_[ops[i].op] = static_cast<uint8_t>(i);
253 }
254 }
255
256 // 1. Check if op_type is in the batch
257 // 2. If it is, run the setup function in the context of the API call (NOT in
258 // the call party).
259 // 3. This setup function returns a promise factory which we'll then run *in*
260 // the party to do initial setup, and have it return the promise that we'll
261 // ultimately poll on til completion.
262 // Once we express our surface API in terms of core internal types this whole
263 // dance will go away.
264 template <grpc_op_type op_type, typename SetupFn>
265 auto OpHandler(SetupFn setup) {
266 using SetupResult = decltype(std::declval<SetupFn>()(grpc_op()));
267 using Impl = OpHandlerImpl<SetupResult, op_type>;
268 if (const grpc_op* op = this->op(op_type)) {
269 auto r = setup(*op);
270 return Impl(std::move(r));
271 } else {
272 return Impl();
273 }
274 }
275
276 const grpc_op* op(grpc_op_type op_type) const {
277 return idxs_[op_type] == 255 ? nullptr : &ops_[idxs_[op_type]];
278 }
279
280 bool has_op(grpc_op_type op_type) const { return idxs_[op_type] != 255; }
281
282 private:
283 const grpc_op* const ops_;
284 std::array<uint8_t, 8> idxs_{255, 255, 255, 255, 255, 255, 255, 255};
285};

287// Defines a promise that calls grpc_cq_end_op() (on first poll) and then waits
288// for the callback supplied to grpc_cq_end_op() to be called, before resolving
289// to Empty{}
290class WaitForCqEndOp {
291 public:
292 WaitForCqEndOp(bool is_closure, void* tag, grpc_error_handle error,
293 grpc_completion_queue* cq)
294 : state_{NotStarted{is_closure, tag, std::move(error), cq}} {}
295
296 Poll<Empty> operator()();
297
298 WaitForCqEndOp(const WaitForCqEndOp&) = delete;
299 WaitForCqEndOp& operator=(const WaitForCqEndOp&) = delete;
300 WaitForCqEndOp(WaitForCqEndOp&& other) noexcept
301 : state_(std::move(std::get<NotStarted>(other.state_))) {
302 other.state_.emplace<Invalid>();
303 }
304 WaitForCqEndOp& operator=(WaitForCqEndOp&& other) noexcept {
305 state_ = std::move(std::get<NotStarted>(other.state_));
306 other.state_.emplace<Invalid>();
307 return *this;
308 }
309
310 private:
311 struct NotStarted {
312 bool is_closure;
313 void* tag;
314 grpc_error_handle error;
315 grpc_completion_queue* cq;
316 };
317 struct Started {
318 explicit Started(Waker waker) : waker(std::move(waker)) {}
319 Waker waker;
320 grpc_cq_completion completion;
321 std::atomic<bool> done{false};
322 };
323 struct Invalid {};
324 using State = std::variant<NotStarted, Started, Invalid>;
325
326 static std::string StateString(const State& state);
327
328 State state_{Invalid{}};
329};

331template <typename FalliblePart, typename FinalPart>
332auto InfallibleBatch(FalliblePart fallible_part, FinalPart final_part,
333 bool is_notify_tag_closure, void* notify_tag,
334 grpc_completion_queue* cq) {
335 // Perform fallible_part, then final_part, then wait for the
336 // completion queue to be done.
337 // If cancelled, we'll ensure the completion queue is notified.
338 // There's a slight bug here in that if we cancel this promise after
339 // the WaitForCqEndOp we'll double post -- but we don't currently do that.
340 return OnCancelFactory(
341 [fallible_part = std::move(fallible_part),
342 final_part = std::move(final_part), is_notify_tag_closure, notify_tag,
343 cq]() mutable {
344 return LogPollBatch(notify_tag,
345 Seq(std::move(fallible_part), std::move(final_part),
346 [is_notify_tag_closure, notify_tag, cq]() {
347 return WaitForCqEndOp(is_notify_tag_closure,
348 notify_tag,
349 absl::OkStatus(), cq);
350 }));
351 },
352 [cq, notify_tag]() {
353 grpc_cq_end_op(
354 cq, notify_tag, absl::OkStatus(),
355 [](void*, grpc_cq_completion* completion) { delete completion; },
356 nullptr, new grpc_cq_completion);
357 });
358}

360template <typename FalliblePart>
361auto FallibleBatch(FalliblePart fallible_part, bool is_notify_tag_closure,
362 void* notify_tag, grpc_completion_queue* cq) {
363 // Perform fallible_part, then wait for the completion queue to be done.
364 // If cancelled, we'll ensure the completion queue is notified.
365 // There's a slight bug here in that if we cancel this promise after
366 // the WaitForCqEndOp we'll double post -- but we don't currently do that.
367 return OnCancelFactory(
368 [fallible_part = std::move(fallible_part), is_notify_tag_closure,
369 notify_tag, cq]() mutable {
370 return LogPollBatch(
371 notify_tag,
372 Seq(std::move(fallible_part),
373 [is_notify_tag_closure, notify_tag, cq](StatusFlag r) {
374 return WaitForCqEndOp(is_notify_tag_closure, notify_tag,
375 StatusCast<absl::Status>(r), cq);
376 }));
377 },
378 [cq]() {
379 grpc_cq_end_op(
380 cq, nullptr, absl::CancelledError(),
381 [](void*, grpc_cq_completion* completion) { delete completion; },
382 nullptr, new grpc_cq_completion);
383 });
384}

429 template <typename Puller>
430 auto MakeBatchOp(const grpc_op& op, Puller* puller) {
431 CHECK_EQ(recv_message_, nullptr);
432 recv_message_ = op.data.recv_message.recv_message;
433 return [this, puller]() mutable {
434 return Map(puller->PullMessage(),
435 [this](typename Puller::NextMessage msg) {
436 return FinishRecvMessage(std::move(msg));
437 });
438 };
439 }
440
441 private:
442 template <typename NextMessage>
443 StatusFlag FinishRecvMessage(NextMessage result) {
444 if (!result.ok()) {
445 GRPC_TRACE_LOG(call, INFO)
446 << Activity::current()->DebugTag()
447 << "[call] RecvMessage: outstanding_recv "
448 "finishes: received end-of-stream with error";
449 *recv_message_ = nullptr;
450 recv_message_ = nullptr;
451 return Failure{};
452 }
453 if (!result.has_value()) {
454 GRPC_TRACE_LOG(call, INFO) << Activity::current()->DebugTag()
455 << "[call] RecvMessage: outstanding_recv "
456 "finishes: received end-of-stream";
457 *recv_message_ = nullptr;
458 recv_message_ = nullptr;
459 return Success{};
460 }
461 MessageHandle message = result.TakeValue();
462 test_only_last_message_flags_ = message->flags();
463 if ((message->flags() & GRPC_WRITE_INTERNAL_COMPRESS) &&
464 (incoming_compression_algorithm_ != GRPC_COMPRESS_NONE)) {
465 *recv_message_ = grpc_raw_compressed_byte_buffer_create(
466 nullptr, 0, incoming_compression_algorithm_);
467 } else {
468 *recv_message_ = grpc_raw_byte_buffer_create(nullptr, 0);
469 }
470 grpc_slice_buffer_move_into(message->payload()->c_slice_buffer(),
471 &(*recv_message_)->data.raw.slice_buffer);
472 GRPC_TRACE_LOG(call, INFO)
473 << Activity::current()->DebugTag()
474 << "[call] RecvMessage: outstanding_recv "
475 "finishes: received "
476 << (*recv_message_)->data.raw.slice_buffer.length << " byte message";
477 recv_message_ = nullptr;
478 return Success{};
479 }
480
481 grpc_byte_buffer** recv_message_ = nullptr;
482 uint32_t test_only_last_message_flags_ = 0;
483 // Compression algorithm for incoming data
484 grpc_compression_algorithm incoming_compression_algorithm_ =
485 GRPC_COMPRESS_NONE;
486};
