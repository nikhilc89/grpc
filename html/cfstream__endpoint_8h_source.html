102class CFStreamEndpoint : public EventEngine::Endpoint {
103 public:
104 CFStreamEndpoint(std::shared_ptr<CFEventEngine> engine,
105 MemoryAllocator memory_allocator) {
106 impl_ = grpc_core::MakeRefCounted<CFStreamEndpointImpl>(
107 std::move(engine), std::move(memory_allocator));
108 }
109 ~CFStreamEndpoint() override { impl_->Shutdown(); }
110
111 bool Read(absl::AnyInvocable<void(absl::Status)> on_read, SliceBuffer* buffer,
112 ReadArgs args) override {
113 return impl_->Read(std::move(on_read), buffer, std::move(args));
114 }
115 bool Write(absl::AnyInvocable<void(absl::Status)> on_writable,
116 SliceBuffer* data, WriteArgs args) override {
117 return impl_->Write(std::move(on_writable), data, std::move(args));
118 }
119
120 const EventEngine::ResolvedAddress& GetPeerAddress() const override {
121 return impl_->GetPeerAddress();
122 }
123 const EventEngine::ResolvedAddress& GetLocalAddress() const override {
124 return impl_->GetLocalAddress();
125 }
126
127 std::vector<size_t> AllWriteMetrics() override { return {}; }
128 std::optional<absl::string_view> GetMetricName(size_t) override {
129 return std::nullopt;
130 }
131 std::optional<size_t> GetMetricKey(absl::string_view) override {
132 return std::nullopt;
133 }
134
135 public:
136 void Connect(absl::AnyInvocable<void(absl::Status)> on_connect,
137 EventEngine::ResolvedAddress addr) {
138 impl_->Connect(std::move(on_connect), std::move(addr));
139 }
140 bool CancelConnect(absl::Status status) {
141 return impl_->CancelConnect(std::move(status));
142 }
143
144 private:
145 grpc_core::RefCountedPtr<CFStreamEndpointImpl> impl_;
146};
