34class PingInterface {
35 public:
36 struct SendPingArgs {
37 bool ack = false;
38 // RFC9113: PING frames MUST contain 8 octets of opaque data in the frame
39 // payload. A sender can include any value it chooses and use those octets
40 // in any fashion.
41 uint64_t opaque_data = 0;
42 };
43 // TODO(tjagtap) : [PH2][P1] Change the return type of the promises to
44 // Promise<Http2Status> type when that is submitted.
45
46 // Returns a promise that creates and sends a ping frame to the peer.
47 virtual Promise<absl::Status> SendPing(SendPingArgs args) = 0;
48
49 // Returns a promise that triggers a write cycle on the transport.
50 virtual Promise<absl::Status> TriggerWrite() = 0;
51
52 // Returns a promise that handles the ping timeout.
53 virtual Promise<absl::Status> PingTimeout() = 0;
54 virtual ~PingInterface() = default;
55};

128 auto PingTimeout(Duration ping_timeout) {
129 std::shared_ptr<InterActivityLatch<void>> latch =
130 std::make_shared<InterActivityLatch<void>>();
131 auto timeout_cb = [latch]() { latch->Set(); };
132 auto id = ping_callbacks_.OnPingTimeout(ping_timeout, event_engine_.get(),
133 std::move(timeout_cb));
134 DCHECK(id.has_value());
135 VLOG(2) << "Ping timeout of duration: " << ping_timeout
136 << " initiated for ping id: " << *id;
137 return Map(latch->Wait(), [latch](Empty) { return absl::OkStatus(); });
138 }
