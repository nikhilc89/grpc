70 // Returns a promise that resolves to a `absl::Status` indicating the result
71 // of the write operation.
72 //
73 // Concurrent writes are not supported, which means callers should not call
74 // `Write()` before the previous write finishes. Doing that results in
75 // undefined behavior.
76 auto Write(SliceBuffer data, WriteArgs write_args) {
77 GRPC_LATENT_SEE_PARENT_SCOPE("GRPC:Write");
78 // Start write and assert previous write finishes.
79 auto prev = write_state_->state.exchange(WriteState::kWriting,
80 std::memory_order_relaxed);
81 CHECK(prev == WriteState::kIdle);
82 bool completed;
83 if (data.Length() == 0) {
84 completed = true;
85 } else {
86 // TODO(ladynana): Replace this with `SliceBufferCast<>` when it is
87 // available.
88 grpc_slice_buffer_swap(write_state_->buffer.c_slice_buffer(),
89 data.c_slice_buffer());
90 // If `Write()` returns true immediately, the callback will not be called.
91 // We still need to call our callback to pick up the result.
92 write_state_->waker = GetContext<Activity>()->MakeNonOwningWaker();
93 completed = endpoint_->Write(
94 [write_state = write_state_](absl::Status status) {
95 ExecCtx exec_ctx;
96 write_state->Complete(std::move(status));
97 },
98 &write_state_->buffer, std::move(write_args));
99 if (completed) write_state_->waker = Waker();
100 }
101 return If(
102 completed,
103 [this]() {
104 return [write_state = write_state_]() {
105 auto prev = write_state->state.exchange(WriteState::kIdle,
106 std::memory_order_relaxed);
107 CHECK(prev == WriteState::kWriting);
108 return absl::OkStatus();
109 };
110 },
111 GRPC_LATENT_SEE_PROMISE(
112 "DelayedWrite", ([this]() {
113 return [write_state = write_state_]() -> Poll<absl::Status> {
114 // If current write isn't finished return `Pending()`, else
115 // return write result.
116 WriteState::State expected = WriteState::kWritten;
117 if (write_state->state.compare_exchange_strong(
118 expected, WriteState::kIdle, std::memory_order_acquire,
119 std::memory_order_relaxed)) {
120 // State was Written, and we changed it to Idle. We can return
121 // the result.
122 return std::move(write_state->result);
123 }
124 // State was not Written; since we're polling it must be
125 // Writing. Assert that and return Pending.
126 CHECK(expected == WriteState::kWriting);
127 return Pending();
128 };
129 })));
130 }

132 // Returns a promise that resolves to `SliceBuffer` with
133 // `num_bytes` bytes.
134 //
135 // Concurrent reads are not supported, which means callers should not call
136 // `Read()` before the previous read finishes. Doing that results in
137 // undefined behavior.
138 auto Read(size_t num_bytes) {
139 GRPC_LATENT_SEE_PARENT_SCOPE("GRPC:Read");
140 // Assert previous read finishes.
141 CHECK(!read_state_->complete.load(std::memory_order_relaxed));
142 // Should not have pending reads.
143 CHECK(read_state_->pending_buffer.Count() == 0u);
144 bool complete = true;
145 while (read_state_->buffer.Length() < num_bytes) {
146 // Set read args with hinted bytes.
147 grpc_event_engine::experimental::EventEngine::Endpoint::ReadArgs
148 read_args;
149 read_args.set_read_hint_bytes(
150 static_cast<int64_t>(num_bytes - read_state_->buffer.Length()));
151 // If `Read()` returns true immediately, the callback will not be
152 // called.
153 read_state_->waker = GetContext<Activity>()->MakeNonOwningWaker();
154 if (endpoint_->Read(
155 [read_state = read_state_, num_bytes](absl::Status status) {
156 ExecCtx exec_ctx;
157 read_state->Complete(std::move(status), num_bytes);
158 },
159 &read_state_->pending_buffer, std::move(read_args))) {
160 read_state_->waker = Waker();
161 read_state_->pending_buffer.MoveFirstNBytesIntoSliceBuffer(
162 read_state_->pending_buffer.Length(), read_state_->buffer);
163 DCHECK(read_state_->pending_buffer.Count() == 0u);
164 } else {
165 complete = false;
166 break;
167 }
168 }
169 return If(
170 complete,
171 [this, num_bytes]() {
172 SliceBuffer ret;
173 grpc_slice_buffer_move_first_no_inline(
174 read_state_->buffer.c_slice_buffer(), num_bytes,
175 ret.c_slice_buffer());
176 return [ret = std::move(
177 ret)]() mutable -> Poll<absl::StatusOr<SliceBuffer>> {
178 return std::move(ret);
179 };
180 },
181 GRPC_LATENT_SEE_PROMISE(
182 "DelayedRead", ([this, num_bytes]() {
183 return [read_state = read_state_,
184 num_bytes]() -> Poll<absl::StatusOr<SliceBuffer>> {
185 if (!read_state->complete.load(std::memory_order_acquire)) {
186 return Pending();
187 }
188 // If read succeeds, return `SliceBuffer` with `num_bytes`
189 // bytes.
190 if (read_state->result.ok()) {
191 SliceBuffer ret;
192 grpc_slice_buffer_move_first_no_inline(
193 read_state->buffer.c_slice_buffer(), num_bytes,
194 ret.c_slice_buffer());
195 read_state->complete.store(false, std::memory_order_relaxed);
196 return std::move(ret);
197 }
198 read_state->complete.store(false, std::memory_order_relaxed);
199 return std::move(read_state->result);
200 };
201 })));
202 }

204 // Returns a promise that resolves to `Slice` with at least
205 // `num_bytes` bytes which should be less than INT64_MAX bytes.
206 //
207 // Concurrent reads are not supported, which means callers should not call
208 // `ReadSlice()` before the previous read finishes. Doing that results in
209 // undefined behavior.
210 auto ReadSlice(size_t num_bytes) {
211 return Map(Read(num_bytes),
212 [](absl::StatusOr<SliceBuffer> buffer) -> absl::StatusOr<Slice> {
213 if (!buffer.ok()) return buffer.status();
214 return buffer->JoinIntoSlice();
215 });
216 }
217
218 // Returns a promise that resolves to a byte with type `uint8_t`.
219 auto ReadByte() {
220 return Map(ReadSlice(1),
221 [](absl::StatusOr<Slice> slice) -> absl::StatusOr<uint8_t> {
222 if (!slice.ok()) return slice.status();
223 return (*slice)[0];
224 });
225 }

227 // Enables RPC receive coalescing and alignment of memory holding received
228 // RPCs.
229 void EnforceRxMemoryAlignmentAndCoalescing() {
230 auto* chaotic_good_ext = grpc_event_engine::experimental::QueryExtension<
231 grpc_event_engine::experimental::ChaoticGoodExtension>(endpoint_.get());
232 if (chaotic_good_ext != nullptr) {
233 chaotic_good_ext->EnforceRxMemoryAlignment();
234 chaotic_good_ext->EnableRpcReceiveCoalescing();
235 if (read_state_->buffer.Length() == 0) {
236 return;
237 }
238
239 // Copy everything from read_state_->buffer into a single slice and
240 // replace the contents of read_state_->buffer with that slice.
241 grpc_slice slice = grpc_slice_malloc_large(read_state_->buffer.Length());
242 CHECK(reinterpret_cast<uintptr_t>(GRPC_SLICE_START_PTR(slice)) % 64 == 0);
243 size_t ofs = 0;
244 for (size_t i = 0; i < read_state_->buffer.Count(); i++) {
245 memcpy(
246 GRPC_SLICE_START_PTR(slice) + ofs,
247 GRPC_SLICE_START_PTR(
248 read_state_->buffer.c_slice_buffer()->slices[i]),
249 GRPC_SLICE_LENGTH(read_state_->buffer.c_slice_buffer()->slices[i]));
250 ofs +=
251 GRPC_SLICE_LENGTH(read_state_->buffer.c_slice_buffer()->slices[i]);
252 }
253
254 read_state_->buffer.Clear();
255 read_state_->buffer.AppendIndexed(
256 grpc_event_engine::experimental::Slice(slice));
257 DCHECK(read_state_->buffer.Length() == ofs);
258 }
259 }

275 struct ReadState : public RefCounted<ReadState> {
276 std::atomic<bool> complete{false};
277 // Read buffer used for storing successful reads given by
278 // `EventEngine::Endpoint` but not yet requested by the caller.
279 grpc_event_engine::experimental::SliceBuffer buffer;
280 // Buffer used to accept data from `EventEngine::Endpoint`.
281 // Every time after a successful read from `EventEngine::Endpoint`, the data
282 // in this buffer should be appended to `buffer`.
283 grpc_event_engine::experimental::SliceBuffer pending_buffer;
284 // Used for store the result from `EventEngine::Endpoint::Read()`.
285 absl::Status result;
286 Waker waker;
287 // Backing endpoint: we keep this on ReadState as reads will need to
288 // repeatedly read until the target size is hit, and we don't want to access
289 // the main object during this dance (indeed the main object may be
290 // deleted).
291 std::weak_ptr<grpc_event_engine::experimental::EventEngine::Endpoint>
292 endpoint;
293
294 void Complete(absl::Status status, size_t num_bytes_requested);
295 };
296
297 struct WriteState : public RefCounted<WriteState> {
298 enum State : uint8_t {
299 kIdle, // Not writing.
300 kWriting, // Write started, but not completed.
301 kWritten, // Write completed.
302 };
303
304 std::atomic<State> state{kIdle};
305 // Write buffer used for `EventEngine::Endpoint::Write()` to ensure the
306 // memory behind the buffer is not lost.
307 grpc_event_engine::experimental::SliceBuffer buffer;
308 // Used for store the result from `EventEngine::Endpoint::Write()`.
309 absl::Status result;
310 Waker waker;
311
312 void Complete(absl::Status status) {
313 result = std::move(status);
314 auto w = std::move(waker);
315 auto prev = state.exchange(kWritten, std::memory_order_release);
316 // Previous state should be Writing. If we got anything else we've entered
317 // the callback path twice.
318 CHECK(prev == kWriting);
319 w.Wakeup();
320 }
321 };
322
323 RefCountedPtr<WriteState> write_state_ = MakeRefCounted<WriteState>();
324 RefCountedPtr<ReadState> read_state_ = MakeRefCounted<ReadState>();
325};
