46 // BulderScope is used to indicate whether a builder is persistent - these
47 // are builders that are used every time the configuration is built, or
48 // ephemeral - each time the configuration is built these are thrown away.
49 //
50 // Considerations for choosing persistent vs ephemeral:
51 // - For testing we want ephemeral builders, so the next test can throw away
52 // configuration.
53 // - For adapting gRPC to different environments we typically want persistent
54 // builders.
55 // - However, if the adaption should run only once per process, then
56 // ephemeral is better.
57 //
58 // Builders are instantiated in scope order - persistent first, ephemeral
59 // second.
60 enum class BuilderScope {
61 kPersistent,
62 kEphemeral,
63 // Must be last, do not use as a scope.
64 kCount,
65 };

134 // Temporarily replaces core configuration with what is built from the
135 // provided BuildFunc that takes (Builder*) and returns void.
136 // Requires no concurrent Get() be called. Restores current core
137 // configuration when this object is destroyed. The default builder
138 // is not backed up or restored.
139 //
140 // Useful for running multiple tests back to back in the same process
141 // without side effects from previous tests.
142 class WithSubstituteBuilder {
143 public:
144 template <typename BuildFunc>
145 explicit WithSubstituteBuilder(BuildFunc build) {
146 // Build core configuration to replace.
147 Builder builder;
148 build(&builder);
149 CoreConfiguration* p = builder.Build();
150
151 // Backup current core configuration and replace/reset.
152 config_restore_ =
153 CoreConfiguration::config_.exchange(p, std::memory_order_acquire);
154 builders_restore_ =
155 CoreConfiguration::builders_[static_cast<size_t>(
156 BuilderScope::kEphemeral)]
157 .exchange(nullptr, std::memory_order_acquire);
158 }
159
160 ~WithSubstituteBuilder() {
161 // Reset and restore.
162 Reset();
163 CHECK(CoreConfiguration::config_.exchange(
164 config_restore_, std::memory_order_acquire) == nullptr);
165 CHECK(CoreConfiguration::builders_[static_cast<size_t>(
166 BuilderScope::kEphemeral)]
167 .exchange(builders_restore_, std::memory_order_acquire) ==
168 nullptr);
169 }
170
171 private:
172 CoreConfiguration* config_restore_;
173 RegisteredBuilder* builders_restore_;
174 };

178 // Get the core configuration; if it does not exist, create it.
179 static const CoreConfiguration& Get() {
180 CoreConfiguration* p = config_.load(std::memory_order_acquire);
181 if (p != nullptr) {
182 return *p;
183 }
184 return BuildNewAndMaybeSet();
185 }

218 // Helper for tests: Reset the configuration, build a special one, run some
219 // code, and then reset the configuration again.
220 // Templatized to be sure no codegen in normal builds.
221 template <typename BuildFunc, typename RunFunc>
222 static void RunWithSpecialConfiguration(BuildFunc build_configuration,
223 RunFunc code_to_run) {
224 WithSubstituteBuilder builder(build_configuration);
225 code_to_run();
226 }

303template <typename Sink>
304void AbslStringify(Sink& sink, CoreConfiguration::BuilderScope scope) {
305 switch (scope) {
306 case CoreConfiguration::BuilderScope::kPersistent:
307 sink.Append("Persistent");
308 break;
309 case CoreConfiguration::BuilderScope::kEphemeral:
310 sink.Append("Ephemeral");
311 break;
312 case CoreConfiguration::BuilderScope::kCount:
313 sink.Append("Count(");
314 sink.Append(std::to_string(static_cast<size_t>(scope)));
315 sink.Append(")");
316 break;
317 }
318}
