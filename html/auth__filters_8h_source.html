42// Handles calling out to credentials to fill in metadata per call.
43class ClientAuthFilter final : public ImplementChannelFilter<ClientAuthFilter> {
44 public:
45 static const grpc_channel_filter kFilter;
46
47 static absl::string_view TypeName() { return "client-auth-filter"; }
48
49 ClientAuthFilter(
50 RefCountedPtr<grpc_channel_security_connector> security_connector,
51 RefCountedPtr<grpc_auth_context> auth_context);
52
53 static absl::StatusOr<std::unique_ptr<ClientAuthFilter>> Create(
54 const ChannelArgs& args, ChannelFilter::Args);
55
56 private:
57 // These methods are early in the declaration order as auto return type is
58 // needed by Call.
59 auto GetMetadataFromCreds(RefCountedPtr<grpc_call_credentials> creds,
60 ClientMetadataHandle md) {
61 return Map(creds->GetRequestMetadata(std::move(md), &args_),
62 [](absl::StatusOr<ClientMetadataHandle> new_metadata) mutable {
63 if (!new_metadata.ok()) {
64 return absl::StatusOr<ClientMetadataHandle>(
65 MaybeRewriteIllegalStatusCode(new_metadata.status(),
66 "call credentials"));
67 }
68 return new_metadata;
69 });
70 }
71
72 auto GetCallCredsMetadata(ClientMetadataHandle md) {
73 auto creds = GetCallCreds();
74 return If(
75 creds.ok(),
76 [this, &creds, md = std::move(md)]() mutable {
77 return If(
78 *creds != nullptr,
79 [this, &creds, &md]() {
80 return GetMetadataFromCreds(std::move(*creds), std::move(md));
81 },
82 [&md]() {
83 return Immediate(
84 absl::StatusOr<ClientMetadataHandle>(std::move(md)));
85 });
86 },
87 [&creds]() {
88 return Immediate(
89 absl::StatusOr<ClientMetadataHandle>(creds.status()));
90 });
91 }
92
93 public:
94 class Call {
95 public:
96 auto OnClientInitialMetadata(ClientMetadataHandle md,
97 ClientAuthFilter* filter) {
98 filter->InstallContext();
99 auto* host = md->get_pointer(HttpAuthorityMetadata());
100 return AssertResultType<absl::StatusOr<ClientMetadataHandle>>(If(
101 host == nullptr,
102 [&md]() mutable -> absl::StatusOr<ClientMetadataHandle> {
103 return std::move(md);
104 },
105 [filter, host, &md]() mutable {
106 return TrySeq(
107 filter->args_.security_connector->CheckCallHost(
108 host->as_string_view(), filter->args_.auth_context.get()),
109 [filter, md = std::move(md)]() mutable {
110 return filter->GetCallCredsMetadata(std::move(md));
111 });
112 }));
113 }
114 static const inline NoInterceptor OnServerInitialMetadata;
115 static const inline NoInterceptor OnClientToServerMessage;
116 static const inline NoInterceptor OnClientToServerHalfClose;
117 static const inline NoInterceptor OnServerToClientMessage;
118 static const inline NoInterceptor OnServerTrailingMetadata;
119 static const inline NoInterceptor OnFinalize;
120 };
121
122 private:
123 void InstallContext();
124 absl::StatusOr<RefCountedPtr<grpc_call_credentials>> GetCallCreds();
125
126 // Contains refs to security connector and auth context.
127 grpc_call_credentials::GetRequestMetadataArgs args_;
128};
