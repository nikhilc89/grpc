89// base class for all channelz entities
90class BaseNode : public RefCounted<BaseNode> {
91 public:
92 // There are only four high level channelz entities. However, to support
93 // GetTopChannelsRequest, we split the Channel entity into two different
94 // types. All children of BaseNode must be one of these types.
95 enum class EntityType {
96 kTopLevelChannel,
97 kInternalChannel,
98 kSubchannel,
99 kServer,
100 kListenSocket,
101 kSocket,
102 kCall,
103 };
104
105 static absl::string_view EntityTypeString(EntityType type) {
106 switch (type) {
107 case EntityType::kTopLevelChannel:
108 return "top_level_channel";
109 case EntityType::kInternalChannel:
110 return "internal_channel";
111 case EntityType::kSubchannel:
112 return "subchannel";
113 case EntityType::kServer:
114 return "server";
115 case EntityType::kListenSocket:
116 return "listen_socket";
117 case EntityType::kSocket:
118 return "socket";
119 case EntityType::kCall:
120 return "call";
121 }
122 return "unknown";
123 }
124
125 protected:
126 BaseNode(EntityType type, std::string name);
127
128 public:
129 ~BaseNode() override;
130
131 bool HasParent(const BaseNode* parent) const {
132 MutexLock lock(&parent_mu_);
133 return parents_.find(parent) != parents_.end();
134 }
135
136 void AddParent(BaseNode* parent) {
137 MutexLock lock(&parent_mu_);
138 parents_.insert(parent->Ref());
139 }
140
141 void RemoveParent(BaseNode* parent) {
142 MutexLock lock(&parent_mu_);
143 parents_.erase(parent);
144 }
145
146 static absl::string_view ChannelArgName() {
147 return GRPC_ARG_CHANNELZ_CONTAINING_BASE_NODE;
148 }
149 static int ChannelArgsCompare(const BaseNode* a, const BaseNode* b) {
150 return QsortCompare(a, b);
151 }
152
153 // All children must implement this function.
154 virtual Json RenderJson() = 0;
155
156 // Renders the json and returns allocated string that must be freed by the
157 // caller.
158 std::string RenderJsonString();
159
160 EntityType type() const { return type_; }
161 intptr_t uuid() {
162 const intptr_t id = uuid_.load(std::memory_order_relaxed);
163 if (id > 0) return id;
164 return UuidSlow();
165 }
166 const std::string& name() const { return name_; }
167
168 void RunZTrace(absl::string_view name, Timestamp deadline,
169 std::map<std::string, std::string> args,
170 std::shared_ptr<grpc_event_engine::experimental::EventEngine>
171 event_engine,
172 absl::AnyInvocable<void(Json output)> callback);
173 Json::Object AdditionalInfo();
174
175 protected:
176 void PopulateJsonFromDataSources(Json::Object& json);
177
178 private:
179 // to allow the ChannelzRegistry to set uuid_ under its lock.
180 friend class ChannelzRegistry;
181 // allow data source to register/unregister itself
182 friend class DataSource;
183 using ParentSet =
184 absl::flat_hash_set<RefCountedPtr<BaseNode>, RefCountedPtrHash<BaseNode>,
185 RefCountedPtrEq<BaseNode>>;
186
187 intptr_t UuidSlow();
188
189 const EntityType type_;
190 std::atomic<intptr_t> uuid_;
191 std::string name_;
192 Mutex data_sources_mu_;
193 absl::InlinedVector<DataSource*, 3> data_sources_
194 ABSL_GUARDED_BY(data_sources_mu_);
195 BaseNode* prev_;
196 BaseNode* next_;
197 mutable Mutex parent_mu_;
198 ParentSet parents_ ABSL_GUARDED_BY(parent_mu_);
199};

259// This class is a helper class for channelz entities that deal with Channels,
260// Subchannels, and Servers, since those have similar proto definitions.
261// This class has the ability to:
262// - track calls_{started,succeeded,failed}
263// - track last_call_started_timestamp
264// - perform rendering of the above items
265class CallCountingHelper final {
266 public:

268 void RecordCallFailed();
269 void RecordCallSucceeded();

272 return {
273 calls_started_.load(std::memory_order_relaxed),
274 calls_succeeded_.load(std::memory_order_relaxed),
275 calls_failed_.load(std::memory_order_relaxed),
276 last_call_started_cycle_.load(std::memory_order_relaxed),
277 };
278 }
279
280 private:
281 // testing peer friend.
282 friend class testing::CallCountingHelperPeer;
283
284 std::atomic<int64_t> calls_started_{0};
285 std::atomic<int64_t> calls_succeeded_{0};
286 std::atomic<int64_t> calls_failed_{0};
287 std::atomic<gpr_cycle_counter> last_call_started_cycle_{0};
288};
