33// A wrapped fd that integrates with the EventEngine poller of the current
34// platform. A GrpcPolledFd knows how to create grpc platform-specific poller
35// handle from "ares_socket_t" sockets, and then sign up for
36// readability/writeability with that poller handle, and do shutdown and
37// destruction.
38class GrpcPolledFd {
39 public:
40 virtual ~GrpcPolledFd() {}
41 // Called when c-ares library is interested and there's no pending callback
42 virtual void RegisterForOnReadableLocked(
43 absl::AnyInvocable<void(absl::Status)> read_closure) = 0;
44 // Called when c-ares library is interested and there's no pending callback
45 virtual void RegisterForOnWriteableLocked(
46 absl::AnyInvocable<void(absl::Status)> write_closure) = 0;
47 // Indicates if there is data left even after just being read from
48 virtual bool IsFdStillReadableLocked() = 0;
49 // Called once and only once. Must cause cancellation of any pending
50 // read/write callbacks. Return true when the Shutdown is confirmed, false
51 // otherwise.
52 //
53 // TODO(yijiem): On Posix, ShutdownLocked will always succeed. On Windows,
54 // ShutdownLocked only succeeds when error is Cancelled. We could remove these
55 // requirements if we changed the FdNode lifetime model so that:
56 // 1. FdNodes and their underlying socket handles remain alive for
57 // the lifetime of the resolver.
58 // 2. Orphaning the resolver triggers shutdown and subsequent cleanup for
59 // all FdNodes and socket handles.
60 GRPC_MUST_USE_RESULT virtual bool ShutdownLocked(absl::Status error) = 0;
61 // Get the underlying ares_socket_t that this was created from
62 virtual ares_socket_t GetWrappedAresSocketLocked() = 0;
63 // A unique name, for logging
64 virtual const char* GetName() const = 0;
65};
66
67// A GrpcPolledFdFactory is 1-to-1 with and owned by a GrpcAresRequest. It knows
68// how to create GrpcPolledFd's for the current platform, and the
69// GrpcAresRequest uses it for all of its fd's.
70class GrpcPolledFdFactory {
71 public:
72 virtual ~GrpcPolledFdFactory() {}
73 // Optionally initializes the GrpcPolledFdFactory with a grpc_core::Mutex*
74 // for synchronization between the AresResolver and the GrpcPolledFds. The
75 // Windows implementation overrides this.
76 virtual void Initialize(grpc_core::Mutex* mutex,
77 EventEngine* event_engine) = 0;
78 // Creates a new wrapped fd for the current platform
79 virtual std::unique_ptr<GrpcPolledFd> NewGrpcPolledFdLocked(
80 ares_socket_t as) = 0;
81 // Optionally configures the ares channel after creation
82 virtual void ConfigureAresChannelLocked(ares_channel channel) = 0;
83};
