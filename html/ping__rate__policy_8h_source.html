30// How many pings do we allow to be inflight at any given time?
31// In older versions of gRPC this was implicitly 1.
32// With the multiping experiment we allow this to rise to 100 by default.
33// TODO(ctiller): consider making this public API
34#define GRPC_ARG_HTTP2_MAX_INFLIGHT_PINGS "grpc.http2.max_inflight_pings"
35
36class Chttp2PingRatePolicy {
37 public:
38 explicit Chttp2PingRatePolicy(const ChannelArgs& args, bool is_client);
39
40 static void SetDefaults(const ChannelArgs& args);
41
42 struct SendGranted {
43 bool operator==(const SendGranted&) const { return true; }
44 };
45 struct TooManyRecentPings {
46 bool operator==(const TooManyRecentPings&) const { return true; }
47 };
48 struct TooSoon {
49 Duration next_allowed_ping_interval;
50 Timestamp last_ping;
51 Duration wait;
52 bool operator==(const TooSoon& other) const {
53 return next_allowed_ping_interval == other.next_allowed_ping_interval &&
54 last_ping == other.last_ping && wait == other.wait;
55 }
56 };
57 using RequestSendPingResult =
58 std::variant<SendGranted, TooManyRecentPings, TooSoon>;
59
60 // Request that one ping be sent.
61 // Returns:
62 // - SendGranted if a ping can be sent.
63 // - TooManyRecentPings if too many pings have been sent recently and we
64 // should wait for some future write.
65 // - TooSoon if we should wait for some time before sending the ping.
66 RequestSendPingResult RequestSendPing(Duration next_allowed_ping_interval,
67 size_t inflight_pings) const;
68 // Notify the policy that one ping has been sent.
69 void SentPing();
70 // Notify the policy that some data has been sent and so we should no longer
71 // block pings on that basis.
72 void ResetPingsBeforeDataRequired();
73 // Notify the policy that we've received some data.
74 void ReceivedDataFrame();
75 std::string GetDebugString() const;
76
77 int TestOnlyMaxPingsWithoutData() const {
78 return max_pings_without_data_sent_;
79 }
80
81 private:
82 const int max_pings_without_data_sent_;
83 const int max_inflight_pings_;
84 int pings_before_data_sending_required_ = 0;
85 Timestamp last_ping_sent_time_ = Timestamp::InfPast();
86};
