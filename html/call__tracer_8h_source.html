51// The base class for all tracer implementations.
52class CallTracerAnnotationInterface {
53 public:
54 // Enum associated with types of Annotations.
55 enum class AnnotationType {
56 kMetadataSizes,
57 kHttpTransport,
58 kDoNotUse_MustBeLast,
59 };
60
61 // Base class to define a new type of annotation.
62 class Annotation {
63 public:
64 explicit Annotation(AnnotationType type) : type_(type) {}
65 AnnotationType type() const { return type_; }
66 virtual std::string ToString() const = 0;
67 virtual ~Annotation() = default;
68
69 private:
70 const AnnotationType type_;
71 };
72
73 virtual ~CallTracerAnnotationInterface() {}
74 // Records an annotation on the call attempt.
75 // TODO(yashykt): If needed, extend this to attach attributes with
76 // annotations.
77 virtual void RecordAnnotation(absl::string_view annotation) = 0;
78 virtual void RecordAnnotation(const Annotation& annotation) = 0;
79 virtual std::string TraceId() = 0;
80 virtual std::string SpanId() = 0;
81 virtual bool IsSampled() = 0;
82 // Indicates whether this tracer is a delegating tracer or not.
83 // `DelegatingClientCallTracer`, `DelegatingClientCallAttemptTracer` and
84 // `DelegatingServerCallTracer` are the only delegating call tracers.
85 virtual bool IsDelegatingTracer() { return false; }
86};

88// The base class for CallAttemptTracer and ServerCallTracer.
89// TODO(yashykt): What's a better name for this?
90class CallTracerInterface : public CallTracerAnnotationInterface {
91 public:
92 ~CallTracerInterface() override {}
93 // Please refer to `grpc_transport_stream_op_batch_payload` for details on
94 // arguments.
95 virtual void RecordSendInitialMetadata(
96 grpc_metadata_batch* send_initial_metadata) = 0;
97 virtual void RecordSendTrailingMetadata(
98 grpc_metadata_batch* send_trailing_metadata) = 0;
99 virtual void RecordSendMessage(const Message& send_message) = 0;
100 // Only invoked if message was actually compressed.
101 virtual void RecordSendCompressedMessage(
102 const Message& send_compressed_message) = 0;
103 // The `RecordReceivedInitialMetadata()` and `RecordReceivedMessage()`
104 // methods should only be invoked when the metadata/message was
105 // successfully received, i.e., without any error.
106 virtual void RecordReceivedInitialMetadata(
107 grpc_metadata_batch* recv_initial_metadata) = 0;
108 virtual void RecordReceivedMessage(const Message& recv_message) = 0;
109 // Only invoked if message was actually decompressed.
110 virtual void RecordReceivedDecompressedMessage(
111 const Message& recv_decompressed_message) = 0;
112 virtual void RecordCancel(grpc_error_handle cancel_error) = 0;

114 struct TransportByteSize {
115 uint64_t framing_bytes = 0;
116 uint64_t data_bytes = 0;
117 uint64_t header_bytes = 0;
118
119 TransportByteSize& operator+=(const TransportByteSize& other);
120 };
121 virtual void RecordIncomingBytes(
122 const TransportByteSize& transport_byte_size) = 0;
123 virtual void RecordOutgoingBytes(
124 const TransportByteSize& transport_byte_size) = 0;

126 // Traces a new TCP transport attempt for this call attempt. Note the TCP
127 // transport may finish tracing and unref the TCP tracer before or after the
128 // call completion in gRPC core. No TCP tracing when null is returned.
129 virtual std::shared_ptr<TcpCallTracer> StartNewTcpTrace() = 0;
130};

132// Interface for a tracer that records activities on a call. Actual attempts for
133// this call are traced with CallAttemptTracer after invoking RecordNewAttempt()
134// on the ClientCallTracer object.
135class ClientCallTracer : public CallTracerAnnotationInterface {
136 public:
137 // Interface for a tracer that records activities on a particular call
138 // attempt.
139 // (A single RPC can have multiple attempts due to retry/hedging policies or
140 // as transparent retry attempts.)
141 class CallAttemptTracer : public CallTracerInterface {
142 public:
143 // Note that not all of the optional label keys are exposed as public API.
144 enum class OptionalLabelKey : std::uint8_t {
145 kXdsServiceName, // not public
146 kXdsServiceNamespace, // not public
147 kLocality,
148 kSize // should be last
149 };
150
151 ~CallAttemptTracer() override {}
152
153 // TODO(yashykt): The following two methods `RecordReceivedTrailingMetadata`
154 // and `RecordEnd` should be moved into CallTracerInterface.
155 // If the call was cancelled before the recv_trailing_metadata op
156 // was started, recv_trailing_metadata and transport_stream_stats
157 // will be null.
158 virtual void RecordReceivedTrailingMetadata(
159 absl::Status status, grpc_metadata_batch* recv_trailing_metadata,
160 // TODO(roth): Remove this argument when the
161 // call_tracer_in_transport experiment finishes rolling out.
162 const grpc_transport_stream_stats* transport_stream_stats) = 0;
163 // Should be the last API call to the object. Once invoked, the tracer
164 // library is free to destroy the object.
165 virtual void RecordEnd() = 0;
166
167 // Sets an optional label on the per-attempt metrics recorded at the end of
168 // the attempt.
169 virtual void SetOptionalLabel(OptionalLabelKey key,
170 RefCountedStringValue value) = 0;
171 };
172
173 ~ClientCallTracer() override {}
174
175 // Records a new attempt for the associated call. \a transparent denotes
176 // whether the attempt is being made as a transparent retry or as a
177 // non-transparent retry/hedging attempt. (There will be at least one attempt
178 // even if the call is not being retried.) The `ClientCallTracer` object
179 // retains ownership to the newly created `CallAttemptTracer` object.
180 // RecordEnd() serves as an indication that the call stack is done with all
181 // API calls, and the tracer library is free to destroy it after that.
182 virtual CallAttemptTracer* StartNewAttempt(bool is_transparent_retry) = 0;
183};

185// Interface for a tracer that records activities on a server call.
186class ServerCallTracer : public CallTracerInterface {
187 public:
188 ~ServerCallTracer() override {}
189 // TODO(yashykt): The following two methods `RecordReceivedTrailingMetadata`
190 // and `RecordEnd` should be moved into CallTracerInterface.
191 virtual void RecordReceivedTrailingMetadata(
192 grpc_metadata_batch* recv_trailing_metadata) = 0;
193 // Should be the last API call to the object. Once invoked, the tracer
194 // library is free to destroy the object.
195 virtual void RecordEnd(const grpc_call_final_info* final_info) = 0;
196};

198// Interface for a factory that can create a ServerCallTracer object per
199// server call.
200class ServerCallTracerFactory {
201 public:
202 struct RawPointerChannelArgTag {};
203
204 virtual ~ServerCallTracerFactory() {}
205
206 virtual ServerCallTracer* CreateNewServerCallTracer(
207 Arena* arena, const ChannelArgs& channel_args) = 0;
208
209 // Returns true if a server is to be traced, false otherwise.
210 virtual bool IsServerTraced(const ChannelArgs& /*args*/) { return true; }
211
212 // Use this method to get the server call tracer factory from channel args,
213 // instead of directly fetching it with `GetObject`.
214 static ServerCallTracerFactory* Get(const ChannelArgs& channel_args);
215
216 // Registers a global ServerCallTracerFactory that will be used by default if
217 // no corresponding channel arg was found. It is only valid to call this
218 // before grpc_init(). It is the responsibility of the caller to maintain
219 // this for the lifetime of the process.
220 static void RegisterGlobal(ServerCallTracerFactory* factory);
221
222 // Deletes any previous registered ServerCallTracerFactory.
223 static void TestOnlyReset();
224
225 static absl::string_view ChannelArgName();
226};
