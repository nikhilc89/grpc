159 // Starts CDS and EDS/DNS watches for the specified cluster if needed.
160 // Adds an entry to cluster_config_map, which will contain the cluster
161 // data if the data is available.
162 // For each EDS cluster, adds the EDS resource to eds_resources_seen.
163 // For each Logical DNS cluster, adds the DNS hostname to dns_names_seen.
164 // For aggregate clusters, calls itself recursively. If leaf_clusters is
165 // non-null, populates it with a list of leaf clusters, or an error if
166 // max depth is exceeded.
167 // Returns true if all resources have been obtained.
168 bool PopulateClusterConfigMap(
169 absl::string_view name, int depth,
170 absl::flat_hash_map<std::string,
171 absl::StatusOr<XdsConfig::ClusterConfig>>*
172 cluster_config_map,
173 std::set<absl::string_view>* eds_resources_seen,
174 std::set<absl::string_view>* dns_names_seen,
175 absl::StatusOr<std::vector<absl::string_view>>* leaf_clusters = nullptr);
