40// Done creates statuses for the end of the iteration. It's templated on the
41// type of the result of the ForEach loop, so that we can introduce new types
42// easily.
43template <typename T>
44struct Done;

46template <>

48 GPR_ATTRIBUTE_ALWAYS_INLINE_FUNCTION static absl::Status Make(

68template <typename T, typename SfinaeVoid = void>
69struct NextValueTraits;
70
71enum class NextValueType {
72 kValue,
73 kEndOfStream,
74 kError,
75};
76
77template <typename T>
78struct NextValueTraits<T, absl::void_t<typename T::value_type>> {
79 using Value = typename T::value_type;
80
81 GPR_ATTRIBUTE_ALWAYS_INLINE_FUNCTION static NextValueType Type(const T& t) {
82 if (t.has_value()) return NextValueType::kValue;
83 if (t.cancelled()) return NextValueType::kError;
84 return NextValueType::kEndOfStream;
85 }
86
87 GPR_ATTRIBUTE_ALWAYS_INLINE_FUNCTION static Value&& TakeValue(T& t) {
88 return std::move(*t);
89 }
90};
91
92template <typename T>
93struct NextValueTraits<ValueOrFailure<std::optional<T>>> {
94 using Value = T;
95
96 GPR_ATTRIBUTE_ALWAYS_INLINE_FUNCTION static NextValueType Type(
97 const ValueOrFailure<std::optional<T>>& t) {
98 if (t.ok()) {
99 if (t.value().has_value()) return NextValueType::kValue;
100 return NextValueType::kEndOfStream;
101 }
102 return NextValueType::kError;
103 }
104
105 GPR_ATTRIBUTE_ALWAYS_INLINE_FUNCTION static Value&& TakeValue(
106 ValueOrFailure<std::optional<T>>& t) {
107 return std::move(**t);
108 }
109};

111template <typename Reader, typename Action>
112class ForEach {
113 private:
114 using ReaderNext = decltype(std::declval<Reader>().Next());
115 using ReaderResult =
116 typename PollTraits<decltype(std::declval<ReaderNext>()())>::Type;
117 using ReaderResultValue = typename NextValueTraits<ReaderResult>::Value;
118 using ActionFactory =
119 promise_detail::RepeatedPromiseFactory<ReaderResultValue, Action>;
120 using ActionPromise = typename ActionFactory::Promise;
121 using ActionResult =
122 typename PollTraits<decltype(std::declval<ActionPromise>()())>::Type;
123
124 public:
125 using Result = decltype(Done<ActionResult>::Make(false));
126
127 GPR_ATTRIBUTE_ALWAYS_INLINE_FUNCTION ForEach(Reader reader, Action action,
128 DebugLocation whence = {})
129 : reader_(std::move(reader)),
130 action_factory_(std::move(action)),
131 whence_(whence) {
132 Construct(&reader_next_, reader_.Next());
133 }
134 GPR_ATTRIBUTE_ALWAYS_INLINE_FUNCTION ~ForEach() {
135 if (reading_next_) {
136 Destruct(&reader_next_);
137 } else {
138 Destruct(&in_action_);
139 }
140 }
141
142 ForEach(const ForEach&) = delete;
143 ForEach& operator=(const ForEach&) = delete;
144 ForEach(ForEach&& other) noexcept
145 : reader_(std::move(other.reader_)),
146 action_factory_(std::move(other.action_factory_)),
147 whence_(other.whence_) {
148 DCHECK(reading_next_);
149 DCHECK(other.reading_next_);
150 Construct(&reader_next_, std::move(other.reader_next_));
151 }
152 ForEach& operator=(ForEach&& other) noexcept {
153 DCHECK(reading_next_);
154 DCHECK(other.reading_next_);
155 reader_ = std::move(other.reader_);
156 action_factory_ = std::move(other.action_factory_);
157 reader_next_ = std::move(other.reader_next_);
158 whence_ = other.whence_;
159 return *this;
160 }
161
162 Poll<Result> operator()() {
163 if (reading_next_) return PollReaderNext();
164 return PollAction();
165 }
166
167 private:
168 struct InAction {
169 InAction(ActionPromise promise, ReaderResult result)
170 : promise(std::move(promise)), result(std::move(result)) {}
171 ActionPromise promise;
172 ReaderResult result;
173 };
174
175 std::string DebugTag() {
176 return absl::StrCat(GetContext<Activity>()->DebugTag(), " FOR_EACH[0x",
177 reinterpret_cast<uintptr_t>(this), "@", whence_.file(),
178 ":", whence_.line(), "]: ");
179 }
180
181 GPR_ATTRIBUTE_ALWAYS_INLINE_FUNCTION Poll<Result> PollReaderNext() {
182 GRPC_TRACE_LOG(promise_primitives, INFO) << DebugTag() << " PollReaderNext";
183 auto r = reader_next_();
184 if (auto* p = r.value_if_ready()) {
185 switch (NextValueTraits<ReaderResult>::Type(*p)) {
186 case NextValueType::kValue: {
187 GRPC_TRACE_LOG(promise_primitives, INFO)
188 << DebugTag() << " PollReaderNext: got value";
189 Destruct(&reader_next_);
190 auto action = action_factory_.Make(
191 NextValueTraits<ReaderResult>::TakeValue(*p));
192 Construct(&in_action_, std::move(action), std::move(*p));
193 reading_next_ = false;
194 return PollAction();
195 }
196 case NextValueType::kEndOfStream: {
197 GRPC_TRACE_LOG(promise_primitives, INFO)
198 << DebugTag() << " PollReaderNext: got end of stream";
199 return Done<Result>::Make(false);
200 }
201 case NextValueType::kError: {
202 GRPC_TRACE_LOG(promise_primitives, INFO)
203 << DebugTag() << " PollReaderNext: got error";
204 return Done<Result>::Make(true);
205 }
206 }
207 }
208 return Pending();
209 }
210
211 Poll<Result> PollAction() {
212 GRPC_TRACE_LOG(promise_primitives, INFO) << DebugTag() << " PollAction";
213 auto r = in_action_.promise();
214 if (auto* p = r.value_if_ready()) {
215 if (IsStatusOk(*p)) {
216 Destruct(&in_action_);
217 Construct(&reader_next_, reader_.Next());
218 reading_next_ = true;
219 return PollReaderNext();
220 } else {
221 return std::move(*p);
222 }
223 }
224 return Pending();
225 }
226
227 GPR_NO_UNIQUE_ADDRESS Reader reader_;
228 GPR_NO_UNIQUE_ADDRESS ActionFactory action_factory_;
229 GPR_NO_UNIQUE_ADDRESS DebugLocation whence_;
230 bool reading_next_ = true;
231 union {
232 ReaderNext reader_next_;
233 InAction in_action_;
234 };
235};
